65// On Windows, MSVC's narrow `argv` is decoded through the C locale
66// (default "C"), which mangles any non-ASCII argument like
67// "/c/Users/Tomáš" into replacement chars. Re-decode from the wide
68// command line to UTF-8 and rewrite argv before anyone reads it.
69#ifdef _WIN32
70static void rewriteArgvAsUtf8(int& argc, char**& argv,
71 std::vector<std::string>& storage,
72 std::vector<char*>& ptrs) {
73 int wargc = 0;
74 LPWSTR* wargv = ::CommandLineToArgvW(::GetCommandLineW(), &wargc);
75 if (!wargv) return;
76 storage.reserve(static_cast<std::size_t>(wargc));
77 for (int i = 0; i < wargc; ++i) {
78 storage.push_back(wbsh::wideToUtf8(wargv[i]));
79 }
80 ::LocalFree(wargv);
81 ptrs.reserve(storage.size() + 1);
82 for (auto& s : storage) ptrs.push_back(s.data());
83 ptrs.push_back(nullptr);
84 argc = static_cast<int>(storage.size());
85 argv = ptrs.data();
86}
87#endif /* _WIN32 */

108static ParseResult parseArgs(int argc, char** argv, CliOptions& opts) {
109 for (int i = 1; i < argc; ++i) {
110 std::string a = argv[i];
111 if (a == "-h" || a == "--help") {
112 printHelp();
113 return { true, 0 };
114 }
115 if (a == "-t" || a == "--tokens") { opts.show_tokens = true; continue; }
116 if (a == "-e" || a == "--expand") { opts.do_expand = true; continue; }
117 if (a == "-r" || a == "--run") { opts.do_run = true; continue; }
118 if (a == "-i" || a == "--interactive") {
119 return { true, runInteractive() };
120 }
121 if (a == "--ast") { opts.show_ast = true; opts.ast_explicit = true; continue; }
122 if (a == "--no-ast") { opts.show_ast = false; opts.ast_explicit = true; continue; }
123 if (a == "-c") {
124 if (i + 1 >= argc) {
125 std::cerr << "wbsh: -c requires an argument\n";
126 return { true, 2 };
127 }
128 opts.src = argv[++i];
129 opts.have_src = true;
130 continue;
131 }
132 if (a == "-") {
133 opts.from_stdin = true;
134 continue;
135 }
136 std::ifstream f(a, std::ios::binary);
137 if (!f) {
138 std::cerr << "wbsh: cannot open file: " << a << "\n";
139 return { true, 2 };
140 }
141 opts.src = readAll(f);
142 opts.have_src = true;
143 }
144 return { false, 0 };
145}

147// Resolve the source-of-input. With no -c / file / `-`, fall through to
148// the interactive REPL when stdin is a TTY, or read piped stdin as a
149// script otherwise.
150static ParseResult resolveSource(CliOptions& opts) {
151 if (opts.have_src) return { false, 0 };
152 if (opts.from_stdin) {
153 opts.src = readAll(std::cin);
154 opts.have_src = true;
155 return { false, 0 };
156 }
157 if (isInteractiveStdin()) {
158 return { true, runInteractive() };
159 }
160 if (!std::cin.eof()) {
161 opts.src = readAll(std::cin);
162 opts.have_src = true;
163 return { false, 0 };
164 }
165 printHelp();
166 return { true, 0 };
167}

169int main(int argc, char** argv) {
170#ifdef _WIN32
171 std::vector<std::string> argv_utf8_storage;
172 std::vector<char*> argv_ptrs;
173 rewriteArgvAsUtf8(argc, argv, argv_utf8_storage, argv_ptrs);
174#endif /* _WIN32 */
175
176 CliOptions opts;
177 if (auto pr = parseArgs(argc, argv, opts); pr.exit_now) return pr.exit_code;
178 if (auto pr = resolveSource(opts); pr.exit_now) return pr.exit_code;
179
180 // In run mode, default to suppressing the AST dump so output is clean.
181 if (opts.do_run && !opts.ast_explicit) opts.show_ast = false;
182 return runOnSource(opts.src, opts.show_tokens, opts.show_ast,
183 opts.do_expand, opts.do_run);
184}
