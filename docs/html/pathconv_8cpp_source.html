104 PathConv::PathConv() {
105 // Drive-letter mounts: /a..z -> A:\ ..Z:\ (one entry per drive letter).
106 for (char c = 'a'; c <= 'z'; ++c) {
107 Mount m;
108 m.posix = std::string("/") + c;
109 std::string w; w += static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
110 w += ":";
111 m.win32 = std::move(w);
112 m.exact = false;
113 mounts_.push_back(std::move(m));
114 }
115
116 // /tmp -> %TEMP%
117 std::string tmp = getTempDir();
118 if (!tmp.empty()) {
119 Mount m{ "/tmp", tmp, false };
120 mounts_.push_back(std::move(m));
121 }
122
123 // /dev/null -> NUL (Windows special device that always exists).
124 mounts_.push_back({ "/dev/null", "NUL", true });
125
126 // Console aliases. /dev/tty resolves to CON, which Win32 supports as
127 // a path that opens the console.
128 mounts_.push_back({ "/dev/tty", "CON", true });
129 mounts_.push_back({ "/dev/stdin", "CONIN$", true });
130 mounts_.push_back({ "/dev/stdout","CONOUT$", true });
131 mounts_.push_back({ "/dev/stderr","CONOUT$", true });
132 }

150 bool PathConv::applyMount(const std::string& p, std::string& out) const {
151 // Find the longest matching mount.
152 const Mount* best = nullptr;
153 std::size_t best_len = 0;
154 for (const auto& m : mounts_) {
155 if (m.exact) {
156 if (p == m.posix) { best = &m; best_len = m.posix.size(); break; }
157 continue;
158 }
159 if (p.size() < m.posix.size()) continue;
160 if (p.compare(0, m.posix.size(), m.posix) != 0) continue;
161 if (p.size() > m.posix.size() && p[m.posix.size()] != '/') continue;
162 if (m.posix.size() > best_len) {
163 best = &m;
164 best_len = m.posix.size();
165 }
166 }
167 if (!best) return false;
168 if (best->exact) {
169 out = best->win32;
170 return true;
171 }
172 // Prefix match. Splice mount.win32 + (rest of p with `/` -> `\`).
173 std::string tail = p.substr(best->posix.size());
174 // Drop leading separator if present (we'll re-add via mount).
175 if (!tail.empty() && tail.front() == '/') tail.erase(0, 1);
176 std::string w = best->win32;
177 if (!tail.empty()) {
178 if (!w.empty() && w.back() != '\\' && w.back() != '/') w.push_back('\\');
179 w += tail;
180 } else {
181 // "/c" alone -> "C:\"
182 if (!w.empty() && w.back() == ':') w.push_back('\\');
183 }
184 slashesToBackslashes(w);
185 out = std::move(w);
186 return true;
187 }

189 std::string PathConv::toWin32(const std::string& p) const {
190 if (p.empty()) return p;
191 // Already a Win32 path? Pass through (just normalise / -> \‍).
192 if (looksLikeWin32(p)) {
193 std::string s = p;
194 slashesToBackslashes(s);
195 return s;
196 }
197 // Special-case bare `/` — root of current drive.
198 if (p == "/") {
199#ifdef _WIN32
200 char buf[MAX_PATH];
201 DWORD n = GetCurrentDirectoryA(MAX_PATH, buf);
202 if (n >= 3 && buf[1] == ':') {
203 return std::string(1, buf[0]) + ":\\";
204 }
205 return "C:\\";
206#else
207 return "/";
208#endif
209 }
210 if (isPosixAbsolute(p)) {
211 std::string out;
212 if (applyMount(p, out)) return out;
213 // Fallback: treat unknown /foo as relative-to-current-drive root.
214 std::string s = p.substr(1);
215 slashesToBackslashes(s);
216#ifdef _WIN32
217 char buf[MAX_PATH];
218 DWORD n = GetCurrentDirectoryA(MAX_PATH, buf);
219 if (n >= 3 && buf[1] == ':') {
220 return std::string(1, buf[0]) + ":\\" + s;
221 }
222#endif
223 return "\\" + s;
224 }
225 // Relative path — leave as-is (forward slashes are accepted by Win32 APIs).
226 return p;
227 }

229 std::string PathConv::toPosix(const std::string& p) const {
230 if (p.empty()) return p;
231 std::string s = p;
232 // Drive-letter prefix: X:\... or X:/... -> /x/...
233 if (s.size() >= 2 && std::isalpha(static_cast<unsigned char>(s[0])) && s[1] == ':') {
234 char drive = static_cast<char>(std::tolower(static_cast<unsigned char>(s[0])));
235 std::string rest = s.substr(2);
236 backslashesToSlashes(rest);
237 if (rest.empty() || rest[0] != '/') rest.insert(rest.begin(), '/');
238 return std::string("/") + drive + rest;
239 }
240 // UNC: \\host\share\... -> //host/share/...
241 if (s.size() >= 2 && (s[0] == '\\' || s[0] == '/') && (s[1] == '\\' || s[1] == '/')) {
242 backslashesToSlashes(s);
243 return s;
244 }
245 // Otherwise convert backslashes only.
246 backslashesToSlashes(s);
247 return s;
248 }

250 std::string PathConv::toWin32Short(const std::string& p) const {
251 std::string longw = toWin32(p);
252#ifdef _WIN32
253 if (longw.empty()) return longw;
254 // Pure-ASCII paths can't be mangled by codepage conversion, so
255 // don't bother hitting the file system.
256 bool has_non_ascii = false;
257 for (unsigned char c : longw) {
258 if (c >= 0x80) { has_non_ascii = true; break; }
259 }
260 if (!has_non_ascii) return longw;
261
262 // UTF-8 -> wide.
263 int wn = ::MultiByteToWideChar(CP_UTF8, 0, longw.data(), (int)longw.size(),
264 nullptr, 0);
265 if (wn <= 0) return longw;
266 std::wstring wlong(static_cast<std::size_t>(wn), L'\0');
267 ::MultiByteToWideChar(CP_UTF8, 0, longw.data(), (int)longw.size(),
268 wlong.data(), wn);
269
270 DWORD need = ::GetShortPathNameW(wlong.c_str(), nullptr, 0);
271 if (need == 0) return longw; // path missing or short names disabled
272 std::wstring wshort(need, L'\0');
273 DWORD got = ::GetShortPathNameW(wlong.c_str(), wshort.data(), need);
274 if (got == 0 || got >= need) return longw;
275 wshort.resize(got);
276
277 // Wide -> UTF-8.
278 int n = ::WideCharToMultiByte(CP_UTF8, 0, wshort.data(), (int)wshort.size(),
279 nullptr, 0, nullptr, nullptr);
280 if (n <= 0) return longw;
281 std::string out(static_cast<std::size_t>(n), '\0');
282 ::WideCharToMultiByte(CP_UTF8, 0, wshort.data(), (int)wshort.size(),
283 out.data(), n, nullptr, nullptr);
284
285 // If 8.3 generation is off on this volume, GetShortPathName returns
286 // the long form unchanged — still non-ASCII, still mangleable.
287 // Fall back to the long form (caller is no worse off than before).
288 for (unsigned char c : out) {
289 if (c >= 0x80) return longw;
290 }
291 return out;
292#else
293 return longw;
294#endif /* _WIN32 */
295 }

316 std::string PathConv::pathListPosixToWin32(const std::string& list) const {
317 if (list.empty()) return list;
318 std::string out;
319 std::string cur;
320 auto flush = [&]() {
321 if (!cur.empty()) {
322 if (!out.empty()) out.push_back(';');
323 out += toWin32(cur);
324 cur.clear();
325 }
326 };
327 // Splitter that treats ':' as separator unless it's a drive-letter colon
328 // (i.e., second char of a token after a single alpha char).
329 std::size_t i = 0;
330 while (i < list.size()) {
331 char c = list[i];
332 if (c == ':') {
333 if (cur.size() == 1 && std::isalpha(static_cast<unsigned char>(cur[0]))) {
334 // Drive-letter colon — part of the entry.
335 cur.push_back(c);
336 } else {
337 flush();
338 }
339 ++i;
340 continue;
341 }
342 cur.push_back(c);
343 ++i;
344 }
345 flush();
346 return out;
347 }

349 bool PathConv::argLooksTranslatable(const std::string& arg) const {
350 if (arg.empty()) return false;
351 if (arg == "/") return false;
352 if (arg.find("://") != std::string::npos) return false;
353 // `--opt=path` and `-x=path` get separately translated by translateArg.
354 if (arg[0] == '-') return false;
355 if (arg[0] != '/') return false;
356 // Single-letter `/x` is almost always a Win32-style switch (`cmd /c`,
357 // `cmd /k`, `where /q`, etc.). Require a deeper path before we
358 // translate, OR an exact match against a known mount.
359 if (arg.size() < 3) return false;
360 // `/x/...` shape is a clear translatable path.
361 if (arg.size() >= 4 && arg[2] == '/') return true;
362 // `/tmp`, `/dev/null`, etc. — exact prefix matches against the
363 // mount table count as translatable.
364 std::string out;
365 return applyMount(arg, out);
366 }

368 std::string PathConv::translateArg(const std::string& arg) const {
369 // Plain absolute path?
370 if (argLooksTranslatable(arg)) return toWin32(arg);
371 // `--opt=PATH` or `-X=PATH` form: translate value.
372 if (!arg.empty() && arg[0] == '-') {
373 std::size_t eq = arg.find('=');
374 if (eq != std::string::npos && eq + 1 < arg.size()) {
375 std::string val = arg.substr(eq + 1);
376 if (argLooksTranslatable(val)) {
377 return arg.substr(0, eq + 1) + toWin32(val);
378 }
379 }
380 }
381 return arg;
382 }
