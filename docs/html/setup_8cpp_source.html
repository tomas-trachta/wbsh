23 // Read a REG_SZ / REG_EXPAND_SZ value as UTF-8, expanding env refs.
24 // Returns empty if the key/value isn't present or the type is wrong.
25 static std::string readRegistryString(HKEY root, const wchar_t* subkey,
26 const wchar_t* value) {
27 HKEY hk;
28 if (RegOpenKeyExW(root, subkey, 0, KEY_QUERY_VALUE, &hk) != ERROR_SUCCESS)
29 return {};
30 DWORD type = 0, size = 0;
31 LONG r = RegQueryValueExW(hk, value, nullptr, &type, nullptr, &size);
32 if (r != ERROR_SUCCESS
33 || (type != REG_SZ && type != REG_EXPAND_SZ)
34 || size == 0) {
35 RegCloseKey(hk);
36 return {};
37 }
38 std::wstring buf(size / sizeof(wchar_t) + 1, L'\0');
39 r = RegQueryValueExW(hk, value, nullptr, &type,
40 reinterpret_cast<LPBYTE>(buf.data()), &size);
41 RegCloseKey(hk);
42 if (r != ERROR_SUCCESS) return {};
43 while (!buf.empty() && buf.back() == L'\0') buf.pop_back();
44 if (type == REG_EXPAND_SZ) {
45 DWORD n = ExpandEnvironmentStringsW(buf.c_str(), nullptr, 0);
46 if (n) {
47 std::wstring exp(n, L'\0');
48 ExpandEnvironmentStringsW(buf.c_str(), exp.data(), n);
49 while (!exp.empty() && exp.back() == L'\0') exp.pop_back();
50 buf = std::move(exp);
51 }
52 }
53 int u = WideCharToMultiByte(CP_UTF8, 0, buf.data(), (int)buf.size(),
54 nullptr, 0, nullptr, nullptr);
55 std::string out(u, '\0');
56 WideCharToMultiByte(CP_UTF8, 0, buf.data(), (int)buf.size(),
57 out.data(), u, nullptr, nullptr);
58 return out;
59 }

61 // Returns the entries of the user + system PATH from the registry, in
62 // Win32 form. Used to repair PATH when wbsh is launched from a parent
63 // (Explorer started at login) whose own environment predates a tool
64 // install. Both cmd.exe and PowerShell get a fresh, registry-merged
65 // PATH from Windows Terminal / login; wbsh, launched from a stale
66 // Explorer process, otherwise wouldn't.
67 static std::vector<std::string> registryPathDirs() {
68 std::vector<std::string> dirs;
69 auto split = [&](const std::string& list) {
70 std::string cur;
71 for (char c : list) {
72 if (c == ';') {
73 if (!cur.empty()) { dirs.push_back(cur); cur.clear(); }
74 } else {
75 cur.push_back(c);
76 }
77 }
78 if (!cur.empty()) dirs.push_back(cur);
79 };
80 // User PATH first so user-installed tools win over system-shadowed ones.
81 split(readRegistryString(HKEY_CURRENT_USER, L"Environment", L"Path"));
82 split(readRegistryString(HKEY_LOCAL_MACHINE,
83 L"SYSTEM\\CurrentControlSet\\Control\\Session Manager\\Environment",
84 L"Path"));
85 return dirs;
86 }

89 // Probe well-known install locations for git.exe and return the
90 // directories that contain it. wbsh ships its own coreutils, but git
91 // itself is an external the user installs separately and absolutely
92 // expects to find on PATH. When wbsh is launched from Explorer the
93 // inherited PATH may miss it; this fills the gap.
94 static std::vector<std::string> findGitDirs() {
95 namespace fs = std::filesystem;
96 std::vector<std::string> candidates;
97 auto add = [&](const char* var, const char* suffix) {
98 if (const char* v = std::getenv(var)) {
99 candidates.push_back(std::string(v) + suffix);
100 }
101 };
102 add("ProgramFiles", "\\Git\\cmd");
103 add("ProgramFiles(x86)", "\\Git\\cmd");
104 add("ProgramW6432", "\\Git\\cmd");
105 add("LOCALAPPDATA", "\\Programs\\Git\\cmd");
106 if (const char* up = std::getenv("USERPROFILE")) {
107 candidates.push_back(std::string(up)
108 + "\\scoop\\apps\\git\\current\\cmd");
109 candidates.push_back(std::string(up)
110 + "\\scoop\\shims");
111 }
112 if (const char* pd = std::getenv("ProgramData")) {
113 candidates.push_back(std::string(pd)
114 + "\\chocolatey\\bin");
115 }
116
117 std::vector<std::string> hits;
118 for (const auto& d : candidates) {
119 std::error_code ec;
120 fs::path exe = fs::path(d) / "git.exe";
121 if (fs::exists(exe, ec) && !fs::is_directory(exe, ec)) {
122 hits.push_back(d);
123 }
124 }
125 return hits;
126 }

164 static void prependDirsToPath(Environment& env, const PathConv& pc,
165 const std::vector<std::string>& dirs) {
166 if (dirs.empty()) return;
167 std::string path = env.get("PATH");
168 std::string prepend;
169 for (const auto& d : dirs) {
170 std::string posix = pc.toPosix(d);
171 // Crude dedup: skip if PATH already contains this entry.
172 if (!path.empty() && path.find(posix) != std::string::npos) continue;
173 if (!prepend.empty()) prepend.push_back(':');
174 prepend += posix;
175 }
176 if (prepend.empty()) return;
177 env.set("PATH", path.empty() ? prepend : (prepend + ":" + path));
178 }

180 void prepareEnv(Environment& env) {
181 env.loadFromProcessEnv();
182 PathConv pc;
183 std::string p = env.get("PATH");
184 if (!p.empty()) env.set("PATH", pc.pathListWin32ToPosix(p));
185#ifdef _WIN32
186 // Merge user + system PATH from the registry. Without this, wbsh
187 // launched from a long-running Explorer can't find tools installed
188 // after that Explorer started -- e.g. VS Code's `code.cmd`.
189 prependDirsToPath(env, pc, registryPathDirs());
190#endif /* _WIN32 */
191 // Make `git` and `docker` discoverable when launched standalone (from
192 // Explorer or via a non-shell parent). Doesn't affect users whose
193 // system PATH already includes them — the dedup check is a no-op there.
194 prependDirsToPath(env, pc, findGitDirs());
195 prependDirsToPath(env, pc, findDockerDirs());
196 std::string home = env.get("HOME");
197 if (home.empty()) home = env.get("USERPROFILE");
198 if (!home.empty()) {
199 env.set("HOME", pc.toPosix(home));
200 env.exportVar("HOME");
201 }
202 std::error_code ec;
203 auto cwd = std::filesystem::current_path(ec);
204 if (!ec) {
205 env.set("PWD", pc.toPosix(pathToUtf8(cwd)));
206 env.exportVar("PWD");
207 }
208 }

210 void absorbInheritedState(Environment& env, Executor& exec) {
211 // Drop any local-name marker from the parent (we already loaded all
212 // values; we just need to fix isExported() before any new spawn).
213 std::string locals_marker = env.get("WBSH_LOCAL_NAMES");
214 if (!locals_marker.empty()) {
215 env.unset("WBSH_LOCAL_NAMES");
216 std::size_t i = 0;
217 while (i < locals_marker.size()) {
218 while (i < locals_marker.size() && locals_marker[i] == ' ') ++i;
219 std::size_t start = i;
220 while (i < locals_marker.size() && locals_marker[i] != ' ') ++i;
221 if (i > start) env.unexportVar(locals_marker.substr(start, i - start));
222 }
223 }
224 std::string inherited_fns = env.get("WBSH_FUNCTIONS");
225 if (!inherited_fns.empty()) {
226 env.unset("WBSH_FUNCTIONS");
227 exec.executeText(inherited_fns, "<inherited functions>");
228 }
229 std::string inherited_aliases = env.get("WBSH_ALIASES");
230 if (!inherited_aliases.empty()) {
231 env.unset("WBSH_ALIASES");
232 exec.executeText(inherited_aliases, "<inherited aliases>");
233 }
234 }
