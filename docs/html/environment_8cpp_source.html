39 void Environment::set(const std::string& name, std::string value) {
40 // Dynamic special parameters: assignment is a state poke, not a
41 // stored value. Reads always synthesise.
42 if (name == "RANDOM") {
43 try { setRandomSeed(static_cast<unsigned int>(std::stoul(value))); }
44 catch (...) {}
45 return;
46 }
47 if (name == "SECONDS") {
48 try { setSecondsOffset(std::stoll(value)); } catch (...) {}
49 return;
50 }
51 if (name == "LINENO" || name == "BASHPID") {
52 // Read-only dynamic params; ignore writes silently.
53 return;
54 }
55 if (readonly_.count(name)) {
56 std::fprintf(stderr, "wbsh: %s: readonly variable\n", name.c_str());
57 return;
58 }
59 // Plain scalar assignment removes any array form bash-style:
60 // `arr=foo` to a previous indexed array sets index 0 only,
61 // keeping the rest. We keep the simpler interpretation: if
62 // there's an existing indexed array, set element 0; otherwise
63 // normal scalar.
64 auto it = indexed_.find(name);
65 if (it != indexed_.end()) {
66 it->second[0] = std::move(value);
67 return;
68 }
69 assoc_.erase(name);
70 vars_[name] = std::move(value);
71 }

99 void Environment::setIndexedElement(const std::string& name, long long idx,
100 std::string val) {
101 if (readonly_.count(name)) {
102 std::fprintf(stderr, "wbsh: %s: readonly variable\n", name.c_str());
103 return;
104 }
105 // Promote a scalar to an indexed array on first element write.
106 if (assoc_.count(name)) {
107 // Treat assoc-keyed numeric assignment as assoc[key=str(idx)].
108 assoc_[name][std::to_string(idx)] = std::move(val);
109 return;
110 }
111 auto it = indexed_.find(name);
112 if (it == indexed_.end()) {
113 IndexedArray ia;
114 auto sv = vars_.find(name);
115 if (sv != vars_.end()) {
116 ia[0] = std::move(sv->second);
117 vars_.erase(sv);
118 }
119 ia[idx] = std::move(val);
120 indexed_[name] = std::move(ia);
121 return;
122 }
123 it->second[idx] = std::move(val);
124 }

166 std::string Environment::get(const std::string& name) const {
167 auto it = vars_.find(name);
168 if (it != vars_.end()) return it->second;
169 // $arr without subscript means $arr[0].
170 auto ix = indexed_.find(name);
171 if (ix != indexed_.end()) {
172 auto e = ix->second.find(0);
173 return e == ix->second.end() ? std::string() : e->second;
174 }
175 auto as = assoc_.find(name);
176 if (as != assoc_.end()) {
177 auto e = as->second.find("0");
178 return e == as->second.end() ? std::string() : e->second;
179 }
180 return {};
181 }

200 // Classic LCG; mask to 0..32767 to match bash.

219 void Environment::loadFromProcessEnv() {
220#ifdef _WIN32
221 LPWCH block = ::GetEnvironmentStringsW();
222 if (!block) return;
223 for (LPWCH p = block; *p; ) {
224 std::wstring entry = p;
225 p += entry.size() + 1;
226 auto eq = entry.find(L'=');
227 if (eq == std::wstring::npos || eq == 0) continue;
228 std::wstring wname = entry.substr(0, eq);
229 std::wstring wval = entry.substr(eq + 1);
230 // Convert to UTF-8 (best-effort using WideCharToMultiByte).
231 auto toUtf8 = [](const std::wstring& w) -> std::string {
232 if (w.empty()) return {};
233 int n = ::WideCharToMultiByte(CP_UTF8, 0, w.data(), (int)w.size(),
234 nullptr, 0, nullptr, nullptr);
235 std::string s(n, '\0');
236 ::WideCharToMultiByte(CP_UTF8, 0, w.data(), (int)w.size(),
237 s.data(), n, nullptr, nullptr);
238 return s;
239 };
240 std::string name = toUtf8(wname);
241 std::string val = toUtf8(wval);
242 vars_[name] = std::move(val);
243 exported_.insert(std::move(name));
244 }
245 ::FreeEnvironmentStringsW(block);
246#else
247 for (char** p = environ; *p; ++p) {
248 std::string entry = *p;
249 auto eq = entry.find('=');
250 if (eq == std::string::npos || eq == 0) continue;
251 std::string name = entry.substr(0, eq);
252 std::string val = entry.substr(eq + 1);
253 vars_[name] = std::move(val);
254 exported_.insert(std::move(name));
255 }
256#endif /* _WIN32 */
257 }
