62 static void escape(std::ostream& os, const std::string& s) {
63 os << '"';
64 for (char c : s) {
65 switch (c) {
66 case '\n': os << "\\n"; break;
67 case '\t': os << "\\t"; break;
68 case '\r': os << "\\r"; break;
69 case '\\': os << "\\\\"; break;
70 case '"': os << "\\\""; break;
71 default: os << c;
72 }
73 }
74 os << '"';
75 }

77 static void expandSimpleCommand(const SimpleCommand& sc, Expander& exp,
78 Environment& env,
79 std::ostream& os, int depth) {
80 indent(os, depth);
81 os << "SimpleCommand @ " << sc.loc.line << ":" << sc.loc.column << "\n";
82 // Commit assignments to env when this is a "pure" assignment command
83 // (no command words). Mirrors bash's `a=1` (set) vs `a=1 cmd` (per-call).
84 bool commit = sc.words.empty();
85 for (const auto& a : sc.assignments) {
86 indent(os, depth + 1);
87 std::string v;
88 try { v = exp.expandStringValue(a.value); }
89 catch (const ExpandError& e) { os << "<expand error: " << e.what() << ">"; }
90 if (commit) env.set(a.name, v);
91 os << "Assign " << a.name << "=";
92 escape(os, v);
93 os << "\n";
94 }
95 for (const auto& w : sc.words) {
96 indent(os, depth + 1);
97 os << "Word raw=";
98 escape(os, w.raw);
99 os << " ->";
100 try {
101 auto fields = exp.expandWord(w);
102 if (fields.empty()) {
103 os << " (vanished)";
104 }
105 for (const auto& f : fields) {
106 os << ' ';
107 escape(os, f);
108 }
109 } catch (const ExpandError& e) {
110 os << " <expand error: " << e.what() << ">";
111 }
112 os << "\n";
113 }
114 for (const auto& r : sc.redirs) {
115 indent(os, depth + 1);
116 os << "Redir " << redirOpName(r.op);
117 if (r.fd != -1) os << " fd=" << r.fd;
118 os << " target=";
119 try { escape(os, exp.expandStringValue(r.target)); }
120 catch (const ExpandError& e) { os << "<expand error: " << e.what() << ">"; }
121 if (r.op == RedirOp::DLess || r.op == RedirOp::DLessDash) {
122 os << " body=";
123 escape(os, exp.expandHeredoc(r.heredoc_body, r.heredoc_quoted));
124 }
125 os << "\n";
126 }
127 }

129 static void walkAndExpand(const Node& n, Expander& exp, Environment& env,
130 std::ostream& os, int depth) {
131 switch (n.kind) {
132 case Node::Kind::SimpleCommand:
133 expandSimpleCommand(static_cast<const SimpleCommand&>(n), exp, env, os, depth);
134 break;
135 case Node::Kind::Pipeline: {
136 const auto& p = static_cast<const Pipeline&>(n);
137 indent(os, depth); os << "Pipeline" << (p.bang ? " (bang)" : "") << "\n";
138 for (const auto& c : p.commands) walkAndExpand(*c, exp, env, os, depth + 1);
139 break;
140 }
141 case Node::Kind::AndOr: {
142 const auto& a = static_cast<const AndOr&>(n);
143 indent(os, depth); os << (a.op == AndOr::Op::AndIf ? "&&" : "||") << "\n";
144 walkAndExpand(*a.left, exp, env, os, depth + 1);
145 walkAndExpand(*a.right, exp, env, os, depth + 1);
146 break;
147 }
148 case Node::Kind::List: {
149 const auto& l = static_cast<const List&>(n);
150 for (const auto& it : l.items) walkAndExpand(*it.command, exp, env, os, depth);
151 break;
152 }
153 case Node::Kind::BraceGroup: {
154 const auto& bg = static_cast<const BraceGroup&>(n);
155 indent(os, depth); os << "BraceGroup\n";
156 if (bg.body) walkAndExpand(*bg.body, exp, env, os, depth + 1);
157 break;
158 }
159 case Node::Kind::Subshell: {
160 const auto& ss = static_cast<const Subshell&>(n);
161 indent(os, depth); os << "Subshell\n";
162 if (ss.body) walkAndExpand(*ss.body, exp, env, os, depth + 1);
163 break;
164 }
165 case Node::Kind::IfClause: {
166 const auto& ic = static_cast<const IfClause&>(n);
167 indent(os, depth); os << "If\n";
168 for (std::size_t i = 0; i < ic.branches.size(); ++i) {
169 indent(os, depth + 1);
170 os << (i == 0 ? "if-cond:" : "elif-cond:") << "\n";
171 if (ic.branches[i].cond) walkAndExpand(*ic.branches[i].cond, exp, env, os, depth + 2);
172 indent(os, depth + 1); os << "then:\n";
173 if (ic.branches[i].body) walkAndExpand(*ic.branches[i].body, exp, env, os, depth + 2);
174 }
175 if (ic.else_body) {
176 indent(os, depth + 1); os << "else:\n";
177 walkAndExpand(*ic.else_body, exp, env, os, depth + 2);
178 }
179 break;
180 }
181 case Node::Kind::WhileClause: {
182 const auto& w = static_cast<const WhileClause&>(n);
183 indent(os, depth); os << (w.until ? "Until\n" : "While\n");
184 indent(os, depth + 1); os << "cond:\n";
185 if (w.cond) walkAndExpand(*w.cond, exp, env, os, depth + 2);
186 indent(os, depth + 1); os << "do:\n";
187 if (w.body) walkAndExpand(*w.body, exp, env, os, depth + 2);
188 break;
189 }
190 case Node::Kind::ForClause: {
191 const auto& f = static_cast<const ForClause&>(n);
192 indent(os, depth); os << "For " << f.var << "\n";
193 if (f.has_in) {
194 indent(os, depth + 1); os << "items:";
195 for (const auto& w : f.items) {
196 try {
197 auto fields = exp.expandWord(w);
198 for (const auto& s : fields) { os << ' '; escape(os, s); }
199 } catch (const ExpandError& e) {
200 os << " <error: " << e.what() << ">";
201 }
202 }
203 os << "\n";
204 }
205 indent(os, depth + 1); os << "do:\n";
206 if (f.body) walkAndExpand(*f.body, exp, env, os, depth + 2);
207 break;
208 }
209 case Node::Kind::CaseClause: {
210 const auto& c = static_cast<const CaseClause&>(n);
211 indent(os, depth); os << "Case subject=";
212 try { escape(os, exp.expandStringValue(c.subject)); }
213 catch (const ExpandError& e) { os << "<error: " << e.what() << ">"; }
214 os << "\n";
215 for (const auto& it : c.items) {
216 indent(os, depth + 1); os << "patterns:";
217 for (const auto& p : it.patterns) {
218 try { os << ' '; escape(os, exp.expandStringValue(p)); }
219 catch (const ExpandError&) { os << " <err>"; }
220 }
221 os << "\n";
222 if (it.body) {
223 indent(os, depth + 2); os << "body:\n";
224 walkAndExpand(*it.body, exp, env, os, depth + 3);
225 }
226 }
227 break;
228 }
229 case Node::Kind::FunctionDef: {
230 const auto& f = static_cast<const FunctionDef&>(n);
231 indent(os, depth); os << "FunctionDef " << f.name << "\n";
232 if (f.body) walkAndExpand(*f.body, exp, env, os, depth + 1);
233 break;
234 }
235 }

238 int runOnSource(const std::string& src, bool show_tokens, bool show_ast,
239 bool do_expand, bool do_run) {
240 Lexer lex(src);
241 auto tokens = lex.tokenize();
242
243 bool out_color = stdoutIsTty();
244 bool err_color = stderrIsTty();
245 auto header = [&](const char* title) {
246 if (out_color) std::cout << "\x1b[35;1m── " << title << " ──\x1b[0m\n";
247 else std::cout << "--- " << title << " ---\n";
248 };
249 const char* ep = err_color ? "\x1b[31;1m" : "";
250 const char* el = err_color ? "\x1b[33m" : "";
251 const char* er = err_color ? "\x1b[0m" : "";
252
253 if (show_tokens) {
254 header("TOKENS");
255 dumpTokens(std::cout, tokens);
256 }
257 for (const auto& e : lex.errors()) {
258 std::cerr << ep << "lex error" << er << " "
259 << el << e.loc.line << ":" << e.loc.column << er
260 << ": " << e.message << "\n";
261 }
262
263 Parser parser(std::move(tokens), src);
264 auto root = parser.parseProgram();
265 if (show_ast) {
266 header("AST");
267 if (root) dumpAst(std::cout, *root);
268 else std::cout << "(empty)\n";
269 }
270 for (const auto& e : parser.errors()) {
271 std::cerr << ep << "parse error" << er << " "
272 << el << e.loc.line << ":" << e.loc.column << er
273 << ": " << e.message << "\n";
274 }
275
276 if (do_expand && root) {
277 Environment env;
278 prepareEnv(env);
279 Expander exp(env, /*sub=*/nullptr);
280 header("EXPANSIONS");
281 walkAndExpand(*root, exp, env, std::cout, 0);
282 }
283
284 if (do_run && root) {
285 Environment env;
286 prepareEnv(env);
287 Executor exec(env);
288 exec.setSourceText(src);
289 absorbInheritedState(env, exec);
290 try {
291 return exec.execute(*root);
292 } catch (ShellExit& e) {
293 return e.status;
294 }
295 }
296
297 return (parser.errors().empty() && lex.errors().empty()) ? 0 : 1;
298 }
