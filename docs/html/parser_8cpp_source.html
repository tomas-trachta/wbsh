196 NodePtr Parser::parseList(bool /*top_level*/) {
197 std::size_t start = srcOffsetHere();
198 auto list = std::make_unique<List>();
199 list->loc = peek().loc;
200 skipNewlines();
201 while (!atEnd()) {
202 auto andor = parseAndOr();
203 if (!andor) break;
204 ListItem it;
205 it.command = std::move(andor);
206 bool had_sep = false;
207 if (match(TokKind::Amp)) { it.background = true; had_sep = true; }
208 else if (match(TokKind::Semi)) { had_sep = true; }
209 else if (match(TokKind::Newline)) { had_sep = true; }
210 list->items.push_back(std::move(it));
211 if (!had_sep) break;
212 skipNewlines();
213 }
214 stampSpan(*list, start);
215 return list;
216 }

218 // ---------------------------------------------------------------------------
219 // And-or
220 // ---------------------------------------------------------------------------
221
222 NodePtr Parser::parseAndOr() {
223 std::size_t start = srcOffsetHere();
224 NodePtr left = parsePipeline();
225 if (!left) return nullptr;
226 while (check(TokKind::AndIf) || check(TokKind::OrIf)) {
227 auto op = check(TokKind::AndIf) ? AndOr::Op::AndIf : AndOr::Op::OrIf;
228 advance();
229 skipNewlines();
230 auto right = parsePipeline();
231 if (!right) {
232 error(peek(), "expected pipeline after && / ||");
233 break;
234 }
235 auto ao = std::make_unique<AndOr>();
236 ao->op = op;
237 ao->loc = left->loc;
238 ao->left = std::move(left);
239 ao->right = std::move(right);
240 ao->src_start = start;
241 ao->src_end = srcOffsetEnd();
242 left = std::move(ao);
243 }
244 // If no &&/|| chain, left already has its own span.
245 if (left && left->src_end == 0) left->src_end = srcOffsetEnd();
246 return left;
247 }

253 NodePtr Parser::parsePipeline() {
254 std::size_t start = srcOffsetHere();
255 // `time` is a reserved word that prefixes a pipeline (not a command),
256 // so it must be parsed here. Only treat it as the keyword when the
257 // next token actually begins a command — otherwise `time` alone (or
258 // followed by a redirection / `;`) is just an argv[0].
259 bool timed = false;
260 if (checkReserved("time")) {
261 std::size_t saved = pos_;
262 advance();
263 if (atCommandStart() || checkReserved("!")) {
264 timed = true;
265 } else {
266 pos_ = saved;
267 }
268 }
269 bool bang = matchReserved("!");
270 auto first = parseCommand();
271 if (!first) {
272 if (bang) error(peek(), "expected command after `!`");
273 if (timed) error(peek(), "expected command after `time`");
274 return nullptr;
275 }
276 if (!bang && !timed && !check(TokKind::Pipe) && !check(TokKind::PipeAmp)) {
277 return first;
278 }
279 auto pipe = std::make_unique<Pipeline>();
280 pipe->bang = bang;
281 pipe->timed = timed;
282 pipe->loc = first->loc;
283 pipe->commands.push_back(std::move(first));
284 while (check(TokKind::Pipe) || check(TokKind::PipeAmp)) {
285 bool amp = check(TokKind::PipeAmp);
286 advance();
287 skipNewlines();
288 auto next = parseCommand();
289 if (!next) {
290 error(peek(), "expected command after pipe");
291 break;
292 }
293 pipe->stderr_to_stdout.push_back(amp);
294 pipe->commands.push_back(std::move(next));
295 }
296 stampSpan(*pipe, start);
297 return pipe;
298 }

304 NodePtr Parser::parseCommand() {
305 if (!atCommandStart()) return nullptr;
306
307 if (checkReserved("{")) return parseBraceGroup();
308 if (checkReserved("if")) return parseIf();
309 if (checkReserved("while")) return parseWhileUntil(false);
310 if (checkReserved("until")) return parseWhileUntil(true);
311 if (checkReserved("for")) return parseFor();
312 if (checkReserved("case")) return parseCase();
313 if (checkReserved("[[")) return parseDBracket();
314
315 if (checkReserved("function")) {
316 SourceLoc loc = peek().loc;
317 advance();
318 if (peek().kind != TokKind::Word) {
319 error(peek(), "expected function name after `function`");
320 return nullptr;
321 }
322 std::string name = peek().text;
323 advance();
324 if (match(TokKind::LParen)) {
325 if (!match(TokKind::RParen)) error(peek(), "expected `)`");
326 }
327 return parseFunctionRest(std::move(name), loc);
328 }
329
330 if (check(TokKind::LParen)) return parseSubshell();
331
332 // Function definition: name '(' ')' compound_command
333 if (peek().kind == TokKind::Word
334 && peek(1).kind == TokKind::LParen
335 && peek(2).kind == TokKind::RParen)
336 {
337 std::string name = peek().text;
338 SourceLoc loc = peek().loc;
339 advance(); advance(); advance(); // consume name ( )
340 return parseFunctionRest(std::move(name), loc);
341 }
342
343 return parseSimpleCommand();
344 }

435 NodePtr Parser::parseFor() {
436 std::size_t start = srcOffsetHere();
437 SourceLoc loc = peek().loc;
438 advance(); // consume `for`
439 if (peek().kind != TokKind::Word) {
440 error(peek(), "expected variable name after `for`");
441 return nullptr;
442 }
443 std::string var = peek().text;
444 advance();
445 skipNewlines();
446
447 bool has_in = false;
448 std::vector<Word> items;
449 if (matchReserved("in")) {
450 has_in = true;
451 while (peek().kind == TokKind::Word
452 && !tokenIsReservedTerminator(peek()))
453 {
454 items.push_back(tokenToWord(advance()));
455 }
456 if (!match(TokKind::Semi) && !match(TokKind::Newline)) {
457 if (!checkReserved("do"))
458 error(peek(), "expected `;` or newline after for-in word list");
459 }
460 skipNewlines();
461 }
462 else {
463 // Allow optional separator before `do`.
464 if (match(TokKind::Semi) || match(TokKind::Newline)) {
465 skipNewlines();
466 }
467 }
468 auto body = parseDoGroup();
469 auto node = std::make_unique<ForClause>();
470 node->loc = loc;
471 node->var = std::move(var);
472 node->has_in = has_in;
473 node->items = std::move(items);
474 node->body = std::move(body);
475 Redirection r;
476 while (tryParseRedirection(r)) node->redirs.push_back(std::move(r));
477 stampSpan(*node, start);
478 return node;
479 }

481 NodePtr Parser::parseCase() {
482 std::size_t start = srcOffsetHere();
483 SourceLoc loc = peek().loc;
484 advance(); // consume `case`
485 if (peek().kind != TokKind::Word) {
486 error(peek(), "expected word after `case`");
487 return nullptr;
488 }
489 Word subject = tokenToWord(advance());
490 skipNewlines();
491 expectReserved("in", "expected `in`");
492 skipNewlines();
493
494 auto node = std::make_unique<CaseClause>();
495 node->loc = loc;
496 node->subject = std::move(subject);
497
498 while (!atEnd() && !checkReserved("esac")) {
499 CaseClause::Item item;
500 // Optional opening (
501 match(TokKind::LParen);
502 // Patterns separated by `|`
503 while (peek().kind == TokKind::Word) {
504 item.patterns.push_back(tokenToWord(advance()));
505 if (!match(TokKind::Pipe)) break;
506 }
507 expect(TokKind::RParen, "expected `)` after case pattern(s)");
508 skipNewlines();
509 if (!checkReserved("esac")
510 && !check(TokKind::DSemi)
511 && !check(TokKind::SemiAmp)
512 && !check(TokKind::DSemiAmp))
513 {
514 item.body = parseList(false);
515 }
516 if (match(TokKind::DSemi)) item.term = CaseClause::Term::DSemi;
517 else if (match(TokKind::SemiAmp)) item.term = CaseClause::Term::SemiAmp;
518 else if (match(TokKind::DSemiAmp)) item.term = CaseClause::Term::DSemiAmp;
519 else item.term = CaseClause::Term::DSemi; // implicit at esac
520 skipNewlines();
521 node->items.push_back(std::move(item));
522 }
523 expectReserved("esac", "expected `esac`");
524 Redirection r;
525 while (tryParseRedirection(r)) node->redirs.push_back(std::move(r));
526 stampSpan(*node, start);
527 return node;
528 }

575 std::unique_ptr<DBracketCond::Expr> Parser::parseDBracketPrimary() {
576 // Parenthesised sub-expression.
577 if (check(TokKind::LParen)) {
578 advance();
579 auto inner = parseDBracketExpr();
580 if (!match(TokKind::RParen)) {
581 error(peek(), "expected `)` inside [[ ... ]]");
582 }
583 return inner;
584 }
585 auto e = std::make_unique<DBracketCond::Expr>();
586 e->k = DBracketCond::Expr::K::Prim;
587
588 // Helper: lookahead to detect a unary operator like -f / -d / ...
589 auto isUnaryOp = [](const std::string& s) {
590 if (s.size() != 2 || s[0] != '-') return false;
591 static const char ops[] = "abcdefghknoprstuwxzGLNOSU";
592 for (char c : ops) if (c == s[1]) return true;
593 return false;
594 };
595 auto isBinaryOp = [](const std::string& s) {
596 return s == "==" || s == "!=" || s == "=" || s == "=~"
597 || s == "-eq" || s == "-ne" || s == "-lt" || s == "-le"
598 || s == "-gt" || s == "-ge"
599 || s == "-ef" || s == "-nt" || s == "-ot";
600 };
601
602 // Unary prefix?
603 if (peek().kind == TokKind::Word
604 && peek().segments.size() == 1
605 && peek().segments[0].kind == WordSegment::Kind::Literal
606 && isUnaryOp(peek().segments[0].text)
607 && peek(1).kind == TokKind::Word
608 && !isReserved(peek(1), "]]"))
609 {
610 // But beware: `-f && something` — the next token is the operand,
611 // only treat as unary if the token AFTER the operand is a
612 // connective / closer.
613 const Token& after = peek(2);
614 bool ok = isReserved(after, "]]")
615 || after.kind == TokKind::AndIf
616 || after.kind == TokKind::OrIf
617 || after.kind == TokKind::RParen;
618 if (ok) {
619 e->op = peek().segments[0].text;
620 advance();
621 e->lhs = tokenToWord(advance());
622 return e;
623 }
624 }
625
626 if (peek().kind != TokKind::Word) {
627 error(peek(), "expected operand in [[ ... ]]");
628 return e;
629 }
630 e->lhs = tokenToWord(advance());
631
632 // Binary operator?
633 auto opAsString = [](const Token& t) -> std::string {
634 if (t.kind == TokKind::Less) return "<";
635 if (t.kind == TokKind::Great) return ">";
636 if (t.kind == TokKind::Word
637 && t.segments.size() == 1
638 && t.segments[0].kind == WordSegment::Kind::Literal) {
639 return t.segments[0].text;
640 }
641 return {};
642 };
643 std::string opstr = opAsString(peek());
644 if (!opstr.empty() && (opstr == "<" || opstr == ">" || isBinaryOp(opstr))
645 && !isReserved(peek(), "]]"))
646 {
647 e->op = opstr;
648 advance();
649 if (peek().kind != TokKind::Word) {
650 error(peek(), "expected right operand in [[ ... ]]");
651 return e;
652 }
653 e->rhs = tokenToWord(advance());
654 }
655 // else: single-word truthiness test (op stays "")
656 return e;
657 }

679 NodePtr Parser::parseFunctionRest(std::string name, SourceLoc loc) {
680 // `loc` already points at the start of the function definition. Use
681 // the span up to the current end-of-stream position.
682 std::size_t start = loc.offset;
683 skipNewlines();
684 auto body = parseCommand();
685 if (!body) {
686 error(peek(), "expected function body");
687 return nullptr;
688 }
689 auto fn = std::make_unique<FunctionDef>();
690 fn->loc = loc;
691 fn->name = std::move(name);
692 fn->body = std::move(body);
693 stampSpan(*fn, start);
694 // Snapshot the body slice from source so the executor can serialise
695 // it later, even after the AST is moved or the source is freed.
696 if (fn->body && source_ && !source_->empty()
697 && fn->body->src_end > fn->body->src_start
698 && fn->body->src_end <= source_->size()) {
699 fn->body_text = source_->substr(fn->body->src_start,
700 fn->body->src_end - fn->body->src_start);
701 }
702 return fn;
703 }

709 NodePtr Parser::parseSimpleCommand() {
710 std::size_t start = srcOffsetHere();
711 auto cmd = std::make_unique<SimpleCommand>();
712 cmd->loc = peek().loc;
713 bool seen_word = false;
714
715 while (true) {
716 if (atEnd()) break;
717 // Redirection (with or without leading IO_NUMBER).
718 if (atRedirOp() || peek().kind == TokKind::IoNumber) {
719 Redirection r;
720 if (tryParseRedirection(r)) {
721 cmd->redirs.push_back(std::move(r));
722 continue;
723 }
724 break;
725 }
726 if (peek().kind == TokKind::Word) {
727 if (!seen_word) {
728 Assignment a;
729 if (extractAssignment(peek(), a)) {
730 advance();
731 // `name=(...)` array literal: the empty-value
732 // scalar form was just consumed; if the next
733 // token is `(`, switch to array mode.
734 if (a.value.segments.empty() && !a.has_subscript
735 && peek().kind == TokKind::LParen)
736 {
737 advance(); // consume `(`
738 a.is_array = true;
739 skipNewlines();
740 while (!atEnd() && peek().kind != TokKind::RParen) {
741 if (peek().kind == TokKind::Newline) {
742 advance(); continue;
743 }
744 if (peek().kind != TokKind::Word) {
745 error(peek(), "unexpected token in array literal");
746 break;
747 }
748 // Detect [key]=value form by inspecting the
749 // raw token text. Bracketed key with `=`
750 // somewhere after `]`.
751 Word w = tokenToWord(advance());
752 Assignment::Keyed item;
753 if (!w.raw.empty() && w.raw[0] == '['
754 && !w.segments.empty()
755 && w.segments[0].kind == WordSegment::Kind::Literal)
756 {
757 const std::string& lit = w.segments[0].text;
758 std::size_t close = lit.find(']');
759 if (close != std::string::npos
760 && close + 1 < lit.size()
761 && lit[close + 1] == '=')
762 {
763 std::string key_text = lit.substr(1, close - 1);
764 std::string val_text = lit.substr(close + 2);
765 WordSegment ks;
766 ks.kind = WordSegment::Kind::Literal;
767 ks.text = std::move(key_text);
768 item.key.segments.push_back(std::move(ks));
769 item.has_key = true;
770 // Value is the rest of the first
771 // segment plus any following segments.
772 if (!val_text.empty()) {
773 WordSegment vs;
774 vs.kind = WordSegment::Kind::Literal;
775 vs.text = std::move(val_text);
776 item.value.segments.push_back(std::move(vs));
777 }
778 for (std::size_t k = 1; k < w.segments.size(); ++k) {
779 item.value.segments.push_back(w.segments[k]);
780 }
781 item.value.raw = w.raw;
782 a.keyed_items.push_back(std::move(item));
783 continue;
784 }
785 }
786 Assignment::Keyed unkeyed;
787 unkeyed.value = std::move(w);
788 a.keyed_items.push_back(std::move(unkeyed));
789 }
790 if (!match(TokKind::RParen)) {
791 error(peek(), "expected `)` to close array literal");
792 }
793 }
794 cmd->assignments.push_back(std::move(a));
795 continue;
796 }
797 }
798 cmd->words.push_back(tokenToWord(advance()));
799 seen_word = true;
800 continue;
801 }
802 // Anything else terminates the simple command.
803 break;
804 }
805
806 if (cmd->words.empty() && cmd->assignments.empty() && cmd->redirs.empty()) {
807 return nullptr;
808 }
809 stampSpan(*cmd, start);
810 return cmd;
811 }

813 bool Parser::extractAssignment(const Token& t, Assignment& out) const {
814 if (t.kind != TokKind::Word || t.segments.empty()) return false;
815 const auto& first = t.segments[0];
816 if (first.kind != WordSegment::Kind::Literal) return false;
817 const std::string& s0 = first.text;
818 std::size_t i = 0;
819 if (i >= s0.size()) return false;
820 char c0 = s0[i];
821 if (!(std::isalpha(static_cast<unsigned char>(c0)) || c0 == '_')) return false;
822 ++i;
823 while (i < s0.size()
824 && (std::isalnum(static_cast<unsigned char>(s0[i])) || s0[i] == '_'))
825 ++i;
826 std::size_t name_end = i;
827
828 out.name = s0.substr(0, name_end);
829 out.loc = t.loc;
830 out.value.loc = t.loc;
831
832 auto append_value_seg = [&](WordSegment seg) {
833 out.value.segments.push_back(std::move(seg));
834 };
835
836 // Simple case: `name=...` with `=` in the first literal segment.
837 if (i < s0.size() && s0[i] == '=') {
838 if (i + 1 < s0.size()) {
839 WordSegment seg;
840 seg.kind = WordSegment::Kind::Literal;
841 seg.text = s0.substr(i + 1);
842 append_value_seg(std::move(seg));
843 }
844 for (std::size_t k = 1; k < t.segments.size(); ++k)
845 append_value_seg(t.segments[k]);
846 auto eqpos = t.text.find('=');
847 out.value.raw = (eqpos == std::string::npos)
848 ? std::string()
849 : t.text.substr(eqpos + 1);
850 return true;
851 }
852
853 // Subscripted: `name[...]=...`. The subscript may span multiple
854 // segments (`m[$key]=v` lexes as Lit("m["), SimpleVar("key"),
855 // Lit("]=v")). Walk segments collecting subscript content until we
856 // hit a literal segment containing `]=` (or `]` then `=` in the
857 // next literal).
858 if (i >= s0.size() || s0[i] != '[') return false;
859 std::size_t cur_seg = 0;
860 std::size_t cur_pos = i + 1; // skip the `[`
861 out.has_subscript = true;
862 out.subscript.loc = t.loc;
863 // First literal slice: from cur_pos onward (or until `]`).
864 bool found_close = false;
865 std::size_t close_seg = 0;
866 std::size_t close_pos = 0;
867 // Scan literal segments to find the `]`.
868 for (std::size_t k = 0; k < t.segments.size(); ++k) {
869 const auto& seg = t.segments[k];
870 if (seg.kind != WordSegment::Kind::Literal) continue;
871 std::size_t start = (k == cur_seg) ? cur_pos : 0;
872 auto rb = seg.text.find(']', start);
873 if (rb != std::string::npos) {
874 close_seg = k;
875 close_pos = rb;
876 found_close = true;
877 break;
878 }
879 }
880 if (!found_close) return false;
881 // Verify `]=` (or `]` immediately followed by `=` next).
882 const auto& close_text = t.segments[close_seg].text;
883 if (close_pos + 1 < close_text.size()) {
884 if (close_text[close_pos + 1] != '=') return false;
885 } else {
886 // `]` is the last char of this literal; next segment must start
887 // with `=` (only meaningful if the next segment is literal).
888 if (close_seg + 1 >= t.segments.size()) return false;
889 const auto& nxt = t.segments[close_seg + 1];
890 if (nxt.kind != WordSegment::Kind::Literal) return false;
891 if (nxt.text.empty() || nxt.text[0] != '=') return false;
892 }
893
894 // Build subscript Word from segments[cur_seg..close_seg], trimming
895 // the leading `[` and trailing `]`.
896 for (std::size_t k = cur_seg; k <= close_seg; ++k) {
897 const auto& seg = t.segments[k];
898 if (k == cur_seg && seg.kind == WordSegment::Kind::Literal) {
899 std::string slice = (k == close_seg)
900 ? seg.text.substr(cur_pos, close_pos - cur_pos)
901 : seg.text.substr(cur_pos);
902 if (!slice.empty()) {
903 WordSegment w;
904 w.kind = WordSegment::Kind::Literal;
905 w.text = std::move(slice);
906 out.subscript.segments.push_back(std::move(w));
907 }
908 } else if (k == close_seg && seg.kind == WordSegment::Kind::Literal) {
909 std::string slice = seg.text.substr(0, close_pos);
910 if (!slice.empty()) {
911 WordSegment w;
912 w.kind = WordSegment::Kind::Literal;
913 w.text = std::move(slice);
914 out.subscript.segments.push_back(std::move(w));
915 }
916 } else {
917 out.subscript.segments.push_back(seg);
918 }
919 }
920
921 // Build value Word from after `=`.
922 if (close_pos + 1 < close_text.size()) {
923 // `]=` in same segment; value starts at close_pos + 2.
924 std::string slice = close_text.substr(close_pos + 2);
925 if (!slice.empty()) {
926 WordSegment w;
927 w.kind = WordSegment::Kind::Literal;
928 w.text = std::move(slice);
929 append_value_seg(std::move(w));
930 }
931 for (std::size_t k = close_seg + 1; k < t.segments.size(); ++k)
932 append_value_seg(t.segments[k]);
933 } else {
934 // `]` ends close_seg; next literal segment starts with `=`.
935 const auto& nxt = t.segments[close_seg + 1];
936 std::string slice = nxt.text.substr(1);
937 if (!slice.empty()) {
938 WordSegment w;
939 w.kind = WordSegment::Kind::Literal;
940 w.text = std::move(slice);
941 append_value_seg(std::move(w));
942 }
943 for (std::size_t k = close_seg + 2; k < t.segments.size(); ++k)
944 append_value_seg(t.segments[k]);
945 }
946 auto eqpos = t.text.find("]=");
947 out.value.raw = (eqpos == std::string::npos)
948 ? std::string()
949 : t.text.substr(eqpos + 2);
950 return true;
951 }

957 bool Parser::tryParseRedirection(Redirection& out) {
958 std::size_t saved = pos_;
959 int fd = -1;
960 if (peek().kind == TokKind::IoNumber) {
961 try { fd = std::stoi(peek().text); }
962 catch (...) { fd = -1; }
963 advance();
964 }
965 if (!atRedirOp()) {
966 pos_ = saved;
967 return false;
968 }
969 auto kind = peek().kind;
970 auto map_op = [](TokKind k) {
971 switch (k) {
972 case TokKind::Less: return RedirOp::Less;
973 case TokKind::Great: return RedirOp::Great;
974 case TokKind::DGreat: return RedirOp::DGreat;
975 case TokKind::LessAnd: return RedirOp::LessAnd;
976 case TokKind::GreatAnd: return RedirOp::GreatAnd;
977 case TokKind::LessGreat: return RedirOp::LessGreat;
978 case TokKind::Clobber: return RedirOp::Clobber;
979 case TokKind::AmpGreat: return RedirOp::AmpGreat;
980 case TokKind::AmpDGreat: return RedirOp::AmpDGreat;
981 case TokKind::DLess: return RedirOp::DLess;
982 case TokKind::DLessDash: return RedirOp::DLessDash;
983 case TokKind::TLess: return RedirOp::TLess;
984 default: return RedirOp::Less;
985 }
986 };
987 out.op = map_op(kind);
988 out.fd = fd;
989 advance();
990 if (peek().kind != TokKind::Word) {
991 error(peek(), "expected word after redirection operator");
992 return false;
993 }
994 const Token& t = peek();
995 out.target = tokenToWord(t);
996 if (kind == TokKind::DLess || kind == TokKind::DLessDash) {
997 out.heredoc_body = t.heredoc_body;
998 out.heredoc_quoted = t.heredoc_quoted;
999 }
1000 advance();
1001 return true;
1002 }
