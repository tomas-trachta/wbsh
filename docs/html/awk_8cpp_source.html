49// ----- AwkValue type -----------------------------------------------------------
50
51// awk values are loosely typed: every value is a string with an optional
52// numeric interpretation. We store both forms lazily.
53struct AwkValue {
54 double n = 0.0;
55 std::string s;
56 bool has_n = false;
57 bool has_s = false;
58
59 AwkValue() = default;
60 static AwkValue num(double x) { AwkValue v; v.n = x; v.has_n = true; return v; }
61 static AwkValue str(std::string x) { AwkValue v; v.s = std::move(x); v.has_s = true; return v; }
62
63 double asNumber() const {
64 if (has_n) return n;
65 if (has_s) {
66 try { return std::stod(s); } catch (...) { return 0.0; }
67 }
68 return 0.0;
69 }
70 std::string asString() const {
71 if (has_s) return s;
72 if (!has_n) return std::string();
73 // Format like awk: integers without decimal, otherwise %g.
74 double d = n;
75 if (d == std::floor(d) && std::abs(d) < 1e16) {
76 char buf[32];
77 std::snprintf(buf, sizeof(buf), "%lld", (long long)d);
78 return buf;
79 }
80 char buf[64];
81 std::snprintf(buf, sizeof(buf), "%.6g", d);
82 return buf;
83 }
84 bool truthy() const {
85 // Strings are truthy iff non-empty (awk numeric-coercion is fine
86 // because we treat the string form as authoritative when set).
87 if (has_s) {
88 // If the string looks numeric, use numeric interpretation.
89 if (!s.empty()) {
90 try { return std::stod(s) != 0.0; } catch (...) {}
91 return !s.empty();
92 }
93 return false;
94 }
95 return n != 0.0;
96 }
97};

193struct AwkLex {
194 const std::string& src;
195 std::size_t i = 0;
196 int line = 1;
197 std::vector<AwkTok> pushback; // LIFO stack
198 bool prev_was_value = false;
199 explicit AwkLex(const std::string& s) : src(s) {}
200 void unread(AwkTok t) { pushback.push_back(std::move(t)); }
201
202 bool atEnd() const { return i >= src.size(); }
203 char peekc(std::size_t off = 0) const {
204 return i + off < src.size() ? src[i + off] : '\0';
205 }
206 void advance() {
207 if (i < src.size()) {
208 if (src[i] == '\n') ++line;
209 ++i;
210 }
211 }
212 void skipBlanks() {
213 while (i < src.size()) {
214 char c = src[i];
215 if (c == ' ' || c == '\t') { advance(); continue; }
216 if (c == '\\' && i + 1 < src.size() && src[i + 1] == '\n') {
217 advance(); advance(); continue;
218 }
219 if (c == '#') {
220 while (i < src.size() && src[i] != '\n') advance();
221 continue;
222 }
223 break;
224 }
225 }
226 AwkTok readId() {
227 std::size_t start = i;
228 while (i < src.size()
229 && (std::isalnum((unsigned char)src[i]) || src[i] == '_')) advance();
230 AwkTok t;
231 t.line = line;
232 t.text = src.substr(start, i - start);
233 // Reserved words.
234 static const std::unordered_map<std::string, TK> kw = {
235 { "BEGIN", TK::K_BEGIN }, { "END", TK::K_END },
236 { "if", TK::K_IF }, { "else", TK::K_ELSE },
237 { "while", TK::K_WHILE }, { "do", TK::K_DO },
238 { "for", TK::K_FOR }, { "in", TK::K_IN },
239 { "break", TK::K_BREAK }, { "continue", TK::K_CONTINUE },
240 { "next", TK::K_NEXT }, { "exit", TK::K_EXIT },
241 { "delete", TK::K_DELETE }, { "print", TK::K_PRINT },
242 { "printf", TK::K_PRINTF }, { "function", TK::K_FUNCTION },
243 { "return", TK::K_RETURN }, { "getline", TK::K_GETLINE },
244 };
245 auto it = kw.find(t.text);
246 if (it != kw.end()) {
247 t.kind = it->second;
248 } else {
249 t.kind = TK::ID;
250 }
251 return t;
252 }
253 AwkTok readNumber() {
254 std::size_t start = i;
255 while (i < src.size() && std::isdigit((unsigned char)src[i])) advance();
256 if (i < src.size() && src[i] == '.') {
257 advance();
258 while (i < src.size() && std::isdigit((unsigned char)src[i])) advance();
259 }
260 if (i < src.size() && (src[i] == 'e' || src[i] == 'E')) {
261 advance();
262 if (i < src.size() && (src[i] == '+' || src[i] == '-')) advance();
263 while (i < src.size() && std::isdigit((unsigned char)src[i])) advance();
264 }
265 AwkTok t;
266 t.line = line;
267 t.text = src.substr(start, i - start);
268 try { t.num = std::stod(t.text); } catch (...) { t.num = 0; }
269 t.kind = TK::NUM;
270 return t;
271 }
272 AwkTok readString() {
273 // We're at the opening quote.
274 advance();
275 std::string s;
276 while (i < src.size() && src[i] != '"') {
277 if (src[i] == '\\' && i + 1 < src.size()) {
278 char n = src[i + 1];
279 char out = n;
280 switch (n) {
281 case 'n': out = '\n'; break;
282 case 't': out = '\t'; break;
283 case 'r': out = '\r'; break;
284 case '\\': out = '\\'; break;
285 case '"': out = '"'; break;
286 case '/': out = '/'; break;
287 default: s.push_back('\\'); out = n; break;
288 }
289 s.push_back(out);
290 advance(); advance();
291 continue;
292 }
293 s.push_back(src[i]);
294 advance();
295 }
296 if (i < src.size()) advance(); // closing quote
297 AwkTok t;
298 t.line = line;
299 t.kind = TK::STR;
300 t.text = std::move(s);
301 return t;
302 }
303 AwkTok readRegex() {
304 // We're at the opening '/'. Bash-awk semantics: only treat '/' as
305 // regex when it can't be a division operator. The caller decides.
306 advance();
307 std::string r;
308 while (i < src.size() && src[i] != '/') {
309 if (src[i] == '\\' && i + 1 < src.size()) {
310 r.push_back(src[i]);
311 r.push_back(src[i + 1]);
312 advance(); advance();
313 continue;
314 }
315 r.push_back(src[i]);
316 advance();
317 }
318 if (i < src.size()) advance();
319 AwkTok t;
320 t.line = line;
321 t.kind = TK::REGEX;
322 t.text = std::move(r);
323 return t;
324 }
325 AwkTok next() {
326 if (!pushback.empty()) {
327 AwkTok t = std::move(pushback.back());
328 pushback.pop_back();
329 return t;
330 }
331 skipBlanks();
332 AwkTok t;
333 t.line = line;
334 if (atEnd()) { t.kind = TK::END; return t; }
335 char c = src[i];
336 if (c == '\n') { advance(); t.kind = TK::NEWLINE; prev_was_value = false; return t; }
337 if (c == ';') { advance(); t.kind = TK::SEMI; prev_was_value = false; return t; }
338 if (c == ',') { advance(); t.kind = TK::COMMA; prev_was_value = false; return t; }
339 if (c == '(') { advance(); t.kind = TK::LPAREN; prev_was_value = false; return t; }
340 if (c == ')') { advance(); t.kind = TK::RPAREN; prev_was_value = true; return t; }
341 if (c == '{') { advance(); t.kind = TK::LBRACE; prev_was_value = false; return t; }
342 if (c == '}') { advance(); t.kind = TK::RBRACE; prev_was_value = false; return t; }
343 if (c == '[') { advance(); t.kind = TK::LBRACK; prev_was_value = false; return t; }
344 if (c == ']') { advance(); t.kind = TK::RBRACK; prev_was_value = true; return t; }
345 if (c == '?') { advance(); t.kind = TK::QUESTION; prev_was_value = false; return t; }
346 if (c == ':') { advance(); t.kind = TK::COLON; prev_was_value = false; return t; }
347 if (c == '$') { advance(); t.kind = TK::DOLLAR; prev_was_value = false; return t; }
348 if (c == '"') { t = readString(); prev_was_value = true; return t; }
349 if (std::isdigit((unsigned char)c)
350 || (c == '.' && i + 1 < src.size() && std::isdigit((unsigned char)src[i + 1])))
351 {
352 t = readNumber(); prev_was_value = true; return t;
353 }
354 if (std::isalpha((unsigned char)c) || c == '_') {
355 t = readId();
356 prev_was_value = (t.kind == TK::ID);
357 return t;
358 }
359 if (c == '/') {
360 if (!prev_was_value) {
361 t = readRegex();
362 prev_was_value = false;
363 return t;
364 }
365 advance();
366 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::SLASH_ASSIGN; prev_was_value = false; return t; }
367 t.kind = TK::SLASH; prev_was_value = false; return t;
368 }
369 if (c == '+') {
370 advance();
371 if (i < src.size() && src[i] == '+') { advance(); t.kind = TK::INC; prev_was_value = true; return t; }
372 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::PLUS_ASSIGN; prev_was_value = false; return t; }
373 t.kind = TK::PLUS; prev_was_value = false; return t;
374 }
375 if (c == '-') {
376 advance();
377 if (i < src.size() && src[i] == '-') { advance(); t.kind = TK::DEC; prev_was_value = true; return t; }
378 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::MINUS_ASSIGN; prev_was_value = false; return t; }
379 t.kind = TK::MINUS; prev_was_value = false; return t;
380 }
381 if (c == '*') {
382 advance();
383 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::STAR_ASSIGN; prev_was_value = false; return t; }
384 t.kind = TK::STAR; prev_was_value = false; return t;
385 }
386 if (c == '%') {
387 advance();
388 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::PERCENT_ASSIGN; prev_was_value = false; return t; }
389 t.kind = TK::PERCENT; prev_was_value = false; return t;
390 }
391 if (c == '^') {
392 advance();
393 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::CARET_ASSIGN; prev_was_value = false; return t; }
394 t.kind = TK::CARET; prev_was_value = false; return t;
395 }
396 if (c == '=') {
397 advance();
398 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::EQ; prev_was_value = false; return t; }
399 t.kind = TK::ASSIGN; prev_was_value = false; return t;
400 }
401 if (c == '!') {
402 advance();
403 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::NE; prev_was_value = false; return t; }
404 if (i < src.size() && src[i] == '~') { advance(); t.kind = TK::NMATCH; prev_was_value = false; return t; }
405 t.kind = TK::NOT; prev_was_value = false; return t;
406 }
407 if (c == '<') {
408 advance();
409 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::LE; prev_was_value = false; return t; }
410 t.kind = TK::LT; prev_was_value = false; return t;
411 }
412 if (c == '>') {
413 advance();
414 if (i < src.size() && src[i] == '>') { advance(); t.kind = TK::APPEND; prev_was_value = false; return t; }
415 if (i < src.size() && src[i] == '=') { advance(); t.kind = TK::GE; prev_was_value = false; return t; }
416 t.kind = TK::GT; prev_was_value = false; return t;
417 }
418 if (c == '&') {
419 advance();
420 if (i < src.size() && src[i] == '&') { advance(); t.kind = TK::AND; prev_was_value = false; return t; }
421 t.kind = TK::END; return t; // unsupported
422 }
423 if (c == '|') {
424 advance();
425 if (i < src.size() && src[i] == '|') { advance(); t.kind = TK::OR; prev_was_value = false; return t; }
426 t.kind = TK::PIPE; prev_was_value = false; return t;
427 }
428 if (c == '~') { advance(); t.kind = TK::MATCH; prev_was_value = false; return t; }
429 // Unknown char: skip.
430 advance();
431 return next();
432 }
433 AwkTok& peek() {
434 if (pushback.empty()) {
435 AwkTok t = next();
436 pushback.push_back(std::move(t));
437 }
438 return pushback.back();
439 }
440};

444struct AwkParser {
445 AwkLex lex;
446 std::string error_msg;
447
448 explicit AwkParser(const std::string& s) : lex(s) {}
449
450 [[noreturn]] void err(const std::string& m) {
451 throw std::runtime_error("awk: parse error: " + m
452 + " (line " + std::to_string(lex.peek().line) + ")");
453 }
454
455 void skipTerm() {
456 while (true) {
457 TK k = lex.peek().kind;
458 if (k == TK::SEMI || k == TK::NEWLINE) { lex.next(); continue; }
459 break;
460 }
461 }
462
463 AwkProgram parseAwkProgram() {
464 AwkProgram p;
465 skipTerm();
466 while (lex.peek().kind != TK::END) {
467 AwkRule r;
468 parseAwkPattern(r.pat);
469 if (lex.peek().kind == TK::LBRACE) {
470 r.action = parseBlock();
471 } else {
472 // Default action is `{ print }`.
473 auto blk = std::make_unique<Stmt>();
474 blk->kind = SK::Block;
475 auto pr = std::make_unique<Stmt>();
476 pr->kind = SK::Print;
477 blk->children.push_back(std::move(pr));
478 r.action = std::move(blk);
479 }
480 p.rules.push_back(std::move(r));
481 skipTerm();
482 }
483 return p;
484 }
485
486 void parseAwkPattern(AwkPattern& pat) {
487 if (lex.peek().kind == TK::K_BEGIN) {
488 lex.next(); pat.kind = AwkPattern::Begin; return;
489 }
490 if (lex.peek().kind == TK::K_END) {
491 lex.next(); pat.kind = AwkPattern::End; return;
492 }
493 if (lex.peek().kind == TK::LBRACE) {
494 pat.kind = AwkPattern::Always; return;
495 }
496 // expr [, expr]
497 pat.e1 = parseExpr();
498 if (lex.peek().kind == TK::COMMA) {
499 lex.next();
500 pat.e2 = parseExpr();
501 pat.kind = AwkPattern::Range;
502 } else {
503 pat.kind = AwkPattern::Expr;
504 }
505 }
506
507 AwkStmtPtr parseBlock() {
508 if (lex.peek().kind != TK::LBRACE) err("expected '{'");
509 lex.next();
510 auto blk = std::make_unique<Stmt>();
511 blk->kind = SK::Block;
512 skipTerm();
513 while (lex.peek().kind != TK::RBRACE && lex.peek().kind != TK::END) {
514 blk->children.push_back(parseStmt());
515 skipTerm();
516 }
517 if (lex.peek().kind == TK::RBRACE) lex.next();
518 return blk;
519 }
520
521 AwkStmtPtr parseStmt() {
522 TK k = lex.peek().kind;
523 if (k == TK::LBRACE) return parseBlock();
524 if (k == TK::K_IF) return parseIf();
525 if (k == TK::K_WHILE) return parseWhile();
526 if (k == TK::K_DO) return parseDoWhile();
527 if (k == TK::K_FOR) return parseFor();
528 if (k == TK::K_BREAK) { lex.next(); auto s = std::make_unique<Stmt>(); s->kind = SK::Break; return s; }
529 if (k == TK::K_CONTINUE) { lex.next(); auto s = std::make_unique<Stmt>(); s->kind = SK::Continue; return s; }
530 if (k == TK::K_NEXT) { lex.next(); auto s = std::make_unique<Stmt>(); s->kind = SK::Next; return s; }
531 if (k == TK::K_EXIT) {
532 lex.next();
533 auto s = std::make_unique<Stmt>(); s->kind = SK::Exit;
534 if (lex.peek().kind != TK::SEMI && lex.peek().kind != TK::NEWLINE
535 && lex.peek().kind != TK::RBRACE && lex.peek().kind != TK::END) {
536 s->exprs.push_back(parseExpr());
537 }
538 return s;
539 }
540 if (k == TK::K_DELETE) {
541 lex.next();
542 auto s = std::make_unique<Stmt>(); s->kind = SK::Delete;
543 s->exprs.push_back(parseUnary());
544 return s;
545 }
546 if (k == TK::K_PRINT || k == TK::K_PRINTF) {
547 lex.next();
548 auto s = std::make_unique<Stmt>();
549 s->kind = (k == TK::K_PRINT) ? SK::Print : SK::Printf;
550 // argument list (possibly empty for print)
551 if (lex.peek().kind != TK::SEMI && lex.peek().kind != TK::NEWLINE
552 && lex.peek().kind != TK::RBRACE && lex.peek().kind != TK::END
553 && lex.peek().kind != TK::GT && lex.peek().kind != TK::APPEND
554 && lex.peek().kind != TK::PIPE)
555 {
556 s->exprs.push_back(parseExpr());
557 while (lex.peek().kind == TK::COMMA) {
558 lex.next();
559 s->exprs.push_back(parseExpr());
560 }
561 }
562 if (lex.peek().kind == TK::GT) { lex.next(); s->redir_kind = 1; s->to_file = parseExpr(); }
563 else if (lex.peek().kind == TK::APPEND) { lex.next(); s->redir_kind = 2; s->to_file = parseExpr(); }
564 else if (lex.peek().kind == TK::PIPE) { lex.next(); s->redir_kind = 3; s->to_file = parseExpr(); }
565 return s;
566 }
567 // Expression statement (possibly assignment).
568 auto s = std::make_unique<Stmt>();
569 s->kind = SK::ExprStmt;
570 s->exprs.push_back(parseExpr());
571 return s;
572 }
573
574 AwkStmtPtr parseIf() {
575 lex.next(); // 'if'
576 if (lex.peek().kind != TK::LPAREN) err("expected '(' after if");
577 lex.next();
578 auto cond = parseExpr();
579 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
580 lex.next();
581 skipTerm();
582 auto then_s = parseStmt();
583 AwkStmtPtr else_s;
584 skipTerm();
585 if (lex.peek().kind == TK::K_ELSE) {
586 lex.next(); skipTerm();
587 else_s = parseStmt();
588 }
589 auto s = std::make_unique<Stmt>();
590 s->kind = SK::If;
591 s->exprs.push_back(std::move(cond));
592 s->children.push_back(std::move(then_s));
593 if (else_s) s->children.push_back(std::move(else_s));
594 return s;
595 }
596
597 AwkStmtPtr parseWhile() {
598 lex.next();
599 if (lex.peek().kind != TK::LPAREN) err("expected '('");
600 lex.next();
601 auto cond = parseExpr();
602 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
603 lex.next();
604 skipTerm();
605 auto body = parseStmt();
606 auto s = std::make_unique<Stmt>();
607 s->kind = SK::While;
608 s->exprs.push_back(std::move(cond));
609 s->children.push_back(std::move(body));
610 return s;
611 }
612
613 AwkStmtPtr parseDoWhile() {
614 lex.next();
615 skipTerm();
616 auto body = parseStmt();
617 skipTerm();
618 if (lex.peek().kind != TK::K_WHILE) err("expected 'while' after do-body");
619 lex.next();
620 if (lex.peek().kind != TK::LPAREN) err("expected '('");
621 lex.next();
622 auto cond = parseExpr();
623 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
624 lex.next();
625 auto s = std::make_unique<Stmt>();
626 s->kind = SK::DoWhile;
627 s->exprs.push_back(std::move(cond));
628 s->children.push_back(std::move(body));
629 return s;
630 }
631
632 AwkStmtPtr parseFor() {
633 lex.next(); // 'for'
634 if (lex.peek().kind != TK::LPAREN) err("expected '('");
635 lex.next();
636 // for (k in arr) { ... }
637 if (lex.peek().kind == TK::ID) {
638 AwkTok save_id = lex.next();
639 if (lex.peek().kind == TK::K_IN) {
640 lex.next();
641 AwkTok arr_name;
642 if (lex.peek().kind != TK::ID) err("expected array name in 'for in'");
643 arr_name = lex.next();
644 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
645 lex.next();
646 skipTerm();
647 auto body = parseStmt();
648 auto s = std::make_unique<Stmt>();
649 s->kind = SK::ForIn;
650 s->name1 = save_id.text;
651 s->name2 = arr_name.text;
652 s->children.push_back(std::move(body));
653 return s;
654 }
655 // Not for-in; push the ID back onto the lookahead stack.
656 lex.unread(std::move(save_id));
657 }
658 // for (init ; cond ; step) body
659 AwkExprPtr init, cond, step;
660 if (lex.peek().kind != TK::SEMI) init = parseExpr();
661 if (lex.peek().kind != TK::SEMI) err("expected ';'");
662 lex.next();
663 if (lex.peek().kind != TK::SEMI) cond = parseExpr();
664 if (lex.peek().kind != TK::SEMI) err("expected ';'");
665 lex.next();
666 if (lex.peek().kind != TK::RPAREN) step = parseExpr();
667 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
668 lex.next();
669 skipTerm();
670 auto body = parseStmt();
671 auto s = std::make_unique<Stmt>();
672 s->kind = SK::For;
673 s->init = std::move(init);
674 s->cond = std::move(cond);
675 s->step = std::move(step);
676 s->children.push_back(std::move(body));
677 return s;
678 }

680 // ---- Expressions ----
681 AwkExprPtr parseExpr() { return parseTernary(); }
682
683 AwkExprPtr parseTernary() {
684 auto a = parseLogicOr();
685 if (lex.peek().kind == TK::QUESTION) {
686 lex.next();
687 auto b = parseTernary();
688 if (lex.peek().kind != TK::COLON) err("expected ':'");
689 lex.next();
690 auto c = parseTernary();
691 auto e = std::make_unique<Expr>();
692 e->kind = EK::Ternary;
693 e->a = std::move(a);
694 e->b = std::move(b);
695 e->c = std::move(c);
696 return e;
697 }
698 // Assignments are right-associative; allow lhs op= rhs after ternary.
699 TK k = lex.peek().kind;
700 if (k == TK::ASSIGN || k == TK::PLUS_ASSIGN || k == TK::MINUS_ASSIGN
701 || k == TK::STAR_ASSIGN || k == TK::SLASH_ASSIGN
702 || k == TK::PERCENT_ASSIGN || k == TK::CARET_ASSIGN)
703 {
704 std::string op;
705 switch (k) {
706 case TK::ASSIGN: op = "="; break;
707 case TK::PLUS_ASSIGN: op = "+="; break;
708 case TK::MINUS_ASSIGN: op = "-="; break;
709 case TK::STAR_ASSIGN: op = "*="; break;
710 case TK::SLASH_ASSIGN: op = "/="; break;
711 case TK::PERCENT_ASSIGN: op = "%="; break;
712 case TK::CARET_ASSIGN: op = "^="; break;
713 default: break;
714 }
715 lex.next();
716 auto rhs = parseTernary();
717 auto e = std::make_unique<Expr>();
718 if (a->kind == EK::Field) {
719 e->kind = EK::FieldAssign;
720 e->op = op;
721 e->a = std::move(a->a);
722 e->b = std::move(rhs);
723 } else {
724 e->kind = EK::Assign;
725 e->op = op;
726 e->a = std::move(a);
727 e->b = std::move(rhs);
728 }
729 return e;
730 }
731 return a;
732 }
733
734 AwkExprPtr parseLogicOr() {
735 auto a = parseLogicAnd();
736 while (lex.peek().kind == TK::OR) {
737 lex.next();
738 auto b = parseLogicAnd();
739 auto e = std::make_unique<Expr>();
740 e->kind = EK::Binary; e->op = "||";
741 e->a = std::move(a); e->b = std::move(b);
742 a = std::move(e);
743 }
744 return a;
745 }
746 AwkExprPtr parseLogicAnd() {
747 auto a = parseInTest();
748 while (lex.peek().kind == TK::AND) {
749 lex.next();
750 auto b = parseInTest();
751 auto e = std::make_unique<Expr>();
752 e->kind = EK::Binary; e->op = "&&";
753 e->a = std::move(a); e->b = std::move(b);
754 a = std::move(e);
755 }
756 return a;
757 }
758 AwkExprPtr parseInTest() {
759 auto a = parseMatch();
760 if (lex.peek().kind == TK::K_IN) {
761 lex.next();
762 if (lex.peek().kind != TK::ID) err("expected array name after 'in'");
763 AwkTok n = lex.next();
764 auto e = std::make_unique<Expr>();
765 e->kind = EK::ArrayInTest;
766 e->a = std::move(a);
767 e->name = n.text;
768 return e;
769 }
770 return a;
771 }
772 AwkExprPtr parseMatch() {
773 auto a = parseRel();
774 while (lex.peek().kind == TK::MATCH || lex.peek().kind == TK::NMATCH) {
775 std::string op = lex.peek().kind == TK::MATCH ? "~" : "!~";
776 lex.next();
777 auto b = parseRel();
778 auto e = std::make_unique<Expr>();
779 e->kind = EK::Binary; e->op = op;
780 e->a = std::move(a); e->b = std::move(b);
781 a = std::move(e);
782 }
783 return a;
784 }
785 AwkExprPtr parseRel() {
786 auto a = parseConcat();
787 while (true) {
788 TK k = lex.peek().kind;
789 std::string op;
790 switch (k) {
791 case TK::LT: op = "<"; break;
792 case TK::LE: op = "<="; break;
793 case TK::GT: op = ">"; break;
794 case TK::GE: op = ">="; break;
795 case TK::EQ: op = "=="; break;
796 case TK::NE: op = "!="; break;
797 default: return a;
798 }
799 lex.next();
800 auto b = parseConcat();
801 auto e = std::make_unique<Expr>();
802 e->kind = EK::Binary; e->op = op;
803 e->a = std::move(a); e->b = std::move(b);
804 a = std::move(e);
805 }
806 }
807 bool startsConcat(TK k) {
808 switch (k) {
809 case TK::NUM: case TK::STR: case TK::ID: case TK::DOLLAR:
810 case TK::LPAREN: case TK::NOT: case TK::MINUS: case TK::PLUS:
811 case TK::INC: case TK::DEC: case TK::REGEX:
812 return true;
813 default:
814 return false;
815 }
816 }
817 AwkExprPtr parseConcat() {
818 auto a = parseAdd();
819 while (startsConcat(lex.peek().kind)) {
820 auto b = parseAdd();
821 auto e = std::make_unique<Expr>();
822 e->kind = EK::Binary; e->op = " "; // string concat
823 e->a = std::move(a); e->b = std::move(b);
824 a = std::move(e);
825 }
826 return a;
827 }
828 AwkExprPtr parseAdd() {
829 auto a = parseMul();
830 while (lex.peek().kind == TK::PLUS || lex.peek().kind == TK::MINUS) {
831 std::string op = lex.peek().kind == TK::PLUS ? "+" : "-";
832 lex.next();
833 auto b = parseMul();
834 auto e = std::make_unique<Expr>();
835 e->kind = EK::Binary; e->op = op;
836 e->a = std::move(a); e->b = std::move(b);
837 a = std::move(e);
838 }
839 return a;
840 }
841 AwkExprPtr parseMul() {
842 auto a = parseExp();
843 while (true) {
844 TK k = lex.peek().kind;
845 std::string op;
846 if (k == TK::STAR) op = "*";
847 else if (k == TK::SLASH) op = "/";
848 else if (k == TK::PERCENT) op = "%";
849 else return a;
850 lex.next();
851 auto b = parseExp();
852 auto e = std::make_unique<Expr>();
853 e->kind = EK::Binary; e->op = op;
854 e->a = std::move(a); e->b = std::move(b);
855 a = std::move(e);
856 }
857 }
858 AwkExprPtr parseExp() {
859 auto a = parseUnary();
860 if (lex.peek().kind == TK::CARET) {
861 lex.next();
862 auto b = parseExp(); // right-associative
863 auto e = std::make_unique<Expr>();
864 e->kind = EK::Binary; e->op = "^";
865 e->a = std::move(a); e->b = std::move(b);
866 a = std::move(e);
867 }
868 return a;
869 }
870 AwkExprPtr parseUnary() {
871 TK k = lex.peek().kind;
872 if (k == TK::NOT || k == TK::MINUS || k == TK::PLUS
873 || k == TK::INC || k == TK::DEC)
874 {
875 std::string op;
876 switch (k) {
877 case TK::NOT: op = "!"; break;
878 case TK::MINUS: op = "-"; break;
879 case TK::PLUS: op = "+"; break;
880 case TK::INC: op = "++"; break;
881 case TK::DEC: op = "--"; break;
882 default: break;
883 }
884 lex.next();
885 auto inner = parseUnary();
886 auto e = std::make_unique<Expr>();
887 e->kind = EK::Unary; e->op = op;
888 e->a = std::move(inner);
889 return e;
890 }
891 return parsePostfix();
892 }
893 AwkExprPtr parsePostfix() {
894 auto a = parsePrimary();
895 while (true) {
896 TK k = lex.peek().kind;
897 if (k == TK::INC || k == TK::DEC) {
898 std::string op = (k == TK::INC) ? "++" : "--";
899 lex.next();
900 auto e = std::make_unique<Expr>();
901 e->kind = EK::PostIncDec; e->op = op;
902 e->a = std::move(a);
903 a = std::move(e);
904 continue;
905 }
906 break;
907 }
908 return a;
909 }
910 AwkExprPtr parsePrimary() {
911 AwkTok t = lex.peek();
912 if (t.kind == TK::NUM) {
913 lex.next();
914 auto e = std::make_unique<Expr>();
915 e->kind = EK::Number; e->num_val = t.num;
916 return e;
917 }
918 if (t.kind == TK::STR) {
919 lex.next();
920 auto e = std::make_unique<Expr>();
921 e->kind = EK::String; e->str_val = t.text;
922 return e;
923 }
924 if (t.kind == TK::REGEX) {
925 lex.next();
926 auto e = std::make_unique<Expr>();
927 e->kind = EK::Regex; e->str_val = t.text;
928 return e;
929 }
930 if (t.kind == TK::DOLLAR) {
931 lex.next();
932 auto inner = parseUnary();
933 auto e = std::make_unique<Expr>();
934 e->kind = EK::Field;
935 e->a = std::move(inner);
936 return e;
937 }
938 if (t.kind == TK::LPAREN) {
939 lex.next();
940 auto inner = parseExpr();
941 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
942 lex.next();
943 auto e = std::make_unique<Expr>();
944 e->kind = EK::Group; e->a = std::move(inner);
945 return e;
946 }
947 if (t.kind == TK::K_GETLINE) {
948 lex.next();
949 auto e = std::make_unique<Expr>();
950 e->kind = EK::Getline;
951 // Optional variable target.
952 if (lex.peek().kind == TK::ID) {
953 e->name = lex.next().text;
954 }
955 // Optional `< file`.
956 if (lex.peek().kind == TK::LT) {
957 lex.next();
958 e->b = parseUnary();
959 }
960 return e;
961 }
962 if (t.kind == TK::ID) {
963 lex.next();
964 // Function call?
965 if (lex.peek().kind == TK::LPAREN) {
966 lex.next();
967 auto e = std::make_unique<Expr>();
968 e->kind = EK::Call;
969 e->name = t.text;
970 if (lex.peek().kind != TK::RPAREN) {
971 e->args.push_back(parseExpr());
972 while (lex.peek().kind == TK::COMMA) {
973 lex.next();
974 e->args.push_back(parseExpr());
975 }
976 }
977 if (lex.peek().kind != TK::RPAREN) err("expected ')'");
978 lex.next();
979 return e;
980 }
981 // Array reference?
982 if (lex.peek().kind == TK::LBRACK) {
983 lex.next();
984 std::vector<AwkExprPtr> subs;
985 subs.push_back(parseExpr());
986 while (lex.peek().kind == TK::COMMA) {
987 lex.next();
988 subs.push_back(parseExpr());
989 }
990 if (lex.peek().kind != TK::RBRACK) err("expected ']'");
991 lex.next();
992 auto e = std::make_unique<Expr>();
993 e->kind = EK::ArrayRef;
994 e->name = t.text;
995 e->args = std::move(subs);
996 return e;
997 }
998 auto e = std::make_unique<Expr>();
999 e->kind = EK::Var;
1000 e->name = t.text;
1001 return e;
1002 }
1003 err("unexpected token");
1004 return nullptr;
1005 }

1017 long long FNR = 0;

1019 std::string OFS = " ";
1020 std::string ORS = "\n";

1022 std::string SUBSEP = std::string(1, '\x1c');
1023 bool exiting = false;
1024 int exit_status = 0;

1032 ~AwkInterp() {
1033 for (auto& kv : output_files) if (kv.second) std::fclose(kv.second);
1034 for (auto& kv : input_files) if (kv.second && kv.second != stdin) std::fclose(kv.second);
1035 }

1037 // Convert a string value to a field array using current FS.
1038 void splitRecord() {
1039 fields.clear();
1040 if (FS == " ") {
1041 // Whitespace-FS: any run of spaces/tabs (and leading/trailing
1042 // stripped). Newlines never split here in practice.
1043 std::size_t i = 0, n = record.size();
1044 while (i < n) {
1045 while (i < n && (record[i] == ' ' || record[i] == '\t')) ++i;
1046 if (i >= n) break;
1047 std::size_t s = i;
1048 while (i < n && record[i] != ' ' && record[i] != '\t') ++i;
1049 fields.push_back(record.substr(s, i - s));
1050 }
1051 } else if (FS.size() == 1) {
1052 std::size_t i = 0, n = record.size();
1053 std::string cur;
1054 while (i < n) {
1055 if (record[i] == FS[0]) {
1056 fields.push_back(std::move(cur));
1057 cur.clear();
1058 } else cur.push_back(record[i]);
1059 ++i;
1060 }
1061 fields.push_back(std::move(cur));
1062 } else {
1063 // FS is a regex.
1064 try {
1065 std::regex re(FS);
1066 auto begin = std::sregex_token_iterator(record.begin(), record.end(), re, -1);
1067 auto end = std::sregex_token_iterator();
1068 for (auto it = begin; it != end; ++it) fields.push_back(*it);
1069 } catch (...) {
1070 fields.push_back(record);
1071 }
1072 }
1073 }
1074
1075 void rebuildRecord() {
1076 // $0 is the OFS-joined fields after any field is mutated.
1077 record.clear();
1078 for (std::size_t i = 0; i < fields.size(); ++i) {
1079 if (i) record += OFS;
1080 record += fields[i];
1081 }
1082 vars["NF"] = AwkValue::num((double)fields.size());
1083 }
1084
1085 AwkValue getField(int n) {
1086 if (n == 0) return AwkValue::str(record);
1087 if (n < 0 || (std::size_t)n > fields.size()) return AwkValue::str("");
1088 return AwkValue::str(fields[(std::size_t)n - 1]);
1089 }
1090 void setField(int n, std::string v) {
1091 if (n == 0) {
1092 record = v;
1093 splitRecord();
1094 vars["NF"] = AwkValue::num((double)fields.size());
1095 return;
1096 }
1097 if (n < 0) return;
1098 if ((std::size_t)n > fields.size()) fields.resize((std::size_t)n);
1099 fields[(std::size_t)n - 1] = std::move(v);
1100 rebuildRecord();
1101 }
1102
1103 AwkValue getVar(const std::string& name) {
1104 if (name == "NF") return AwkValue::num((double)fields.size());
1105 if (name == "NR") return AwkValue::num((double)NR);
1106 if (name == "FNR") return AwkValue::num((double)FNR);
1107 if (name == "FS") return AwkValue::str(FS);
1108 if (name == "OFS") return AwkValue::str(OFS);
1109 if (name == "ORS") return AwkValue::str(ORS);
1110 if (name == "FILENAME") return AwkValue::str(FILENAME);
1111 if (name == "SUBSEP") return AwkValue::str(SUBSEP);
1112 auto it = vars.find(name);
1113 if (it == vars.end()) return AwkValue::str("");
1114 return it->second;
1115 }
1116 void setVar(const std::string& name, AwkValue v) {
1117 if (name == "FS") { FS = v.asString(); return; }
1118 if (name == "OFS") { OFS = v.asString(); return; }
1119 if (name == "ORS") { ORS = v.asString(); return; }
1120 if (name == "NF") {
1121 int nf = (int)v.asNumber();
1122 if (nf < 0) nf = 0;
1123 fields.resize((std::size_t)nf);
1124 rebuildRecord();
1125 return;
1126 }
1127 if (name == "NR") { NR = (long long)v.asNumber(); return; }
1128 if (name == "FNR") { FNR = (long long)v.asNumber(); return; }
1129 if (name == "FILENAME") { FILENAME = v.asString(); return; }
1130 if (name == "SUBSEP") { SUBSEP = v.asString(); return; }
1131 vars[name] = std::move(v);
1132 }

1136 for (std::size_t i = 0; i < subs.size(); ++i) {
1137 if (i) key += SUBSEP;

1143 // ---- Expression evaluation ----
1144 AwkValue eval(Expr& e) {
1145 switch (e.kind) {
1146 case EK::Number: return AwkValue::num(e.num_val);
1147 case EK::String: return AwkValue::str(e.str_val);
1148 case EK::Regex: return AwkValue::str(e.str_val);
1149 case EK::Var: return getVar(e.name);
1150 case EK::Field: {
1151 int n = (int)eval(*e.a).asNumber();
1152 return getField(n);
1153 }
1154 case EK::ArrayRef: {
1155 std::string key = buildSubscript(e.args);
1156 auto& m = arrays[e.name];
1157 auto it = m.find(key);
1158 return it == m.end() ? AwkValue::str("") : it->second;
1159 }
1160 case EK::ArrayInTest: {
1161 std::string key = eval(*e.a).asString();
1162 auto& m = arrays[e.name];
1163 return AwkValue::num(m.count(key) ? 1.0 : 0.0);
1164 }
1165 case EK::Group: return eval(*e.a);
1166 case EK::Unary: {
1167 AwkValue v = eval(*e.a);
1168 if (e.op == "!") return AwkValue::num(v.truthy() ? 0.0 : 1.0);
1169 if (e.op == "-") return AwkValue::num(-v.asNumber());
1170 if (e.op == "+") return AwkValue::num(v.asNumber());
1171 if (e.op == "++" || e.op == "--") {
1172 double cur = v.asNumber();
1173 double nv = (e.op == "++") ? cur + 1 : cur - 1;
1174 assignLValue(*e.a, AwkValue::num(nv));
1175 return AwkValue::num(nv);
1176 }
1177 return AwkValue::str("");
1178 }
1179 case EK::PostIncDec: {
1180 double cur = eval(*e.a).asNumber();
1181 double nv = (e.op == "++") ? cur + 1 : cur - 1;
1182 assignLValue(*e.a, AwkValue::num(nv));
1183 return AwkValue::num(cur);
1184 }
1185 case EK::Binary: {
1186 if (e.op == "&&") {
1187 if (!eval(*e.a).truthy()) return AwkValue::num(0.0);
1188 return AwkValue::num(eval(*e.b).truthy() ? 1.0 : 0.0);
1189 }
1190 if (e.op == "||") {
1191 if (eval(*e.a).truthy()) return AwkValue::num(1.0);
1192 return AwkValue::num(eval(*e.b).truthy() ? 1.0 : 0.0);
1193 }
1194 if (e.op == "~" || e.op == "!~") {
1195 std::string s = eval(*e.a).asString();
1196 std::string pat;
1197 if (e.b->kind == EK::Regex) pat = e.b->str_val;
1198 else pat = eval(*e.b).asString();
1199 bool m = false;
1200 try {
1201 std::regex re(pat);
1202 m = std::regex_search(s, re);
1203 } catch (...) {}
1204 bool r = (e.op == "~") ? m : !m;
1205 return AwkValue::num(r ? 1.0 : 0.0);
1206 }
1207 if (e.op == " ") {
1208 std::string a = eval(*e.a).asString();
1209 std::string b = eval(*e.b).asString();
1210 return AwkValue::str(a + b);
1211 }
1212 AwkValue av = eval(*e.a);
1213 AwkValue bv = eval(*e.b);
1214 if (e.op == "+") return AwkValue::num(av.asNumber() + bv.asNumber());
1215 if (e.op == "-") return AwkValue::num(av.asNumber() - bv.asNumber());
1216 if (e.op == "*") return AwkValue::num(av.asNumber() * bv.asNumber());
1217 if (e.op == "/") {
1218 double d = bv.asNumber();
1219 if (d == 0) return AwkValue::num(0);
1220 return AwkValue::num(av.asNumber() / d);
1221 }
1222 if (e.op == "%") {
1223 double d = bv.asNumber();
1224 if (d == 0) return AwkValue::num(0);
1225 return AwkValue::num(std::fmod(av.asNumber(), d));
1226 }
1227 if (e.op == "^") return AwkValue::num(std::pow(av.asNumber(), bv.asNumber()));
1228 if (e.op == "==" || e.op == "!=" || e.op == "<" || e.op == "<="
1229 || e.op == ">" || e.op == ">=")
1230 {
1231 bool numeric = av.has_n || bv.has_n;
1232 bool r;
1233 if (numeric) {
1234 double a = av.asNumber(), b = bv.asNumber();
1235 if (e.op == "==") r = a == b;
1236 else if (e.op == "!=") r = a != b;
1237 else if (e.op == "<") r = a < b;
1238 else if (e.op == "<=") r = a <= b;
1239 else if (e.op == ">") r = a > b;
1240 else r = a >= b;
1241 } else {
1242 std::string a = av.asString(), b = bv.asString();
1243 if (e.op == "==") r = a == b;
1244 else if (e.op == "!=") r = a != b;
1245 else if (e.op == "<") r = a < b;
1246 else if (e.op == "<=") r = a <= b;
1247 else if (e.op == ">") r = a > b;
1248 else r = a >= b;
1249 }
1250 return AwkValue::num(r ? 1.0 : 0.0);
1251 }
1252 return AwkValue::str("");
1253 }
1254 case EK::Ternary: {
1255 AwkValue c = eval(*e.a);
1256 return c.truthy() ? eval(*e.b) : eval(*e.c);
1257 }
1258 case EK::Assign: return assignTo(*e.a, e.op, eval(*e.b));
1259 case EK::FieldAssign: {
1260 int n = (int)eval(*e.a).asNumber();
1261 AwkValue cur = getField(n);
1262 AwkValue rhs = eval(*e.b);
1263 AwkValue out = applyOp(cur, e.op, rhs);
1264 setField(n, out.asString());
1265 return out;
1266 }
1267 case EK::Call: return callBuiltin(e);
1268 case EK::Getline: return doGetline(e);
1269 }
1270 return AwkValue::str("");
1271 }

1273 AwkValue applyOp(const AwkValue& cur, const std::string& op, const AwkValue& rhs) {
1274 if (op == "=") return rhs;
1275 double a = cur.asNumber(), b = rhs.asNumber();
1276 if (op == "+=") return AwkValue::num(a + b);
1277 if (op == "-=") return AwkValue::num(a - b);
1278 if (op == "*=") return AwkValue::num(a * b);
1279 if (op == "/=") return AwkValue::num(b == 0 ? 0 : a / b);
1280 if (op == "%=") return AwkValue::num(b == 0 ? 0 : std::fmod(a, b));
1281 if (op == "^=") return AwkValue::num(std::pow(a, b));
1282 return rhs;
1283 }

1285 AwkValue assignTo(Expr& lhs, const std::string& op, AwkValue rhs) {
1286 if (lhs.kind == EK::Var) {
1287 AwkValue cur = getVar(lhs.name);
1288 AwkValue out = applyOp(cur, op, rhs);
1289 setVar(lhs.name, out);
1290 return out;
1291 }
1292 if (lhs.kind == EK::ArrayRef) {
1293 std::string key = buildSubscript(lhs.args);
1294 auto& m = arrays[lhs.name];
1295 AwkValue cur = m.count(key) ? m[key] : AwkValue::str("");
1296 AwkValue out = applyOp(cur, op, rhs);
1297 m[key] = out;
1298 return out;
1299 }
1300 if (lhs.kind == EK::Field) {
1301 int n = (int)eval(*lhs.a).asNumber();
1302 AwkValue cur = getField(n);
1303 AwkValue out = applyOp(cur, op, rhs);
1304 setField(n, out.asString());
1305 return out;
1306 }
1307 return rhs;
1308 }

1314 AwkValue callBuiltin(Expr& e) {
1315 const std::string& n = e.name;
1316 auto sval = [&](std::size_t i) -> std::string {
1317 return i < e.args.size() ? eval(*e.args[i]).asString() : std::string();
1318 };
1319 auto nval = [&](std::size_t i) -> double {
1320 return i < e.args.size() ? eval(*e.args[i]).asNumber() : 0.0;
1321 };
1322 if (n == "length") {
1323 if (e.args.empty()) return AwkValue::num((double)record.size());
1324 return AwkValue::num((double)sval(0).size());
1325 }
1326 if (n == "substr") {
1327 std::string s = sval(0);
1328 long long start = (long long)nval(1);
1329 if (start < 1) start = 1;
1330 long long len = (e.args.size() >= 3)
1331 ? (long long)nval(2)
1332 : (long long)s.size() - start + 1;
1333 if (start > (long long)s.size() || len <= 0) return AwkValue::str("");
1334 return AwkValue::str(s.substr((std::size_t)(start - 1),
1335 (std::size_t)std::min<long long>(len, (long long)s.size() - start + 1)));
1336 }
1337 if (n == "index") {
1338 auto s = sval(0); auto t = sval(1);
1339 if (t.empty()) return AwkValue::num(0);
1340 auto p = s.find(t);
1341 return AwkValue::num(p == std::string::npos ? 0 : (double)(p + 1));
1342 }
1343 if (n == "split") {
1344 std::string s = sval(0);
1345 std::string sep = (e.args.size() >= 3) ? sval(2) : FS;
1346 std::string aname;
1347 if (e.args.size() >= 2 && e.args[1]->kind == EK::Var) {
1348 aname = e.args[1]->name;
1349 } else if (e.args.size() >= 2 && e.args[1]->kind == EK::ArrayRef) {
1350 aname = e.args[1]->name;
1351 }
1352 arrays[aname].clear();
1353 std::vector<std::string> parts;
1354 if (sep == " ") {
1355 std::size_t i = 0, m = s.size();
1356 while (i < m) {
1357 while (i < m && (s[i] == ' ' || s[i] == '\t')) ++i;
1358 if (i >= m) break;
1359 std::size_t st = i;
1360 while (i < m && s[i] != ' ' && s[i] != '\t') ++i;
1361 parts.push_back(s.substr(st, i - st));
1362 }
1363 } else if (sep.size() == 1) {
1364 std::string cur;
1365 for (char c : s) {
1366 if (c == sep[0]) { parts.push_back(std::move(cur)); cur.clear(); }
1367 else cur.push_back(c);
1368 }
1369 parts.push_back(std::move(cur));
1370 } else {
1371 try {
1372 std::regex re(sep);
1373 auto bg = std::sregex_token_iterator(s.begin(), s.end(), re, -1);
1374 auto ed = std::sregex_token_iterator();
1375 for (auto it = bg; it != ed; ++it) parts.push_back(*it);
1376 } catch (...) { parts.push_back(s); }
1377 }
1378 for (std::size_t i = 0; i < parts.size(); ++i) {
1379 arrays[aname][std::to_string(i + 1)] = AwkValue::str(parts[i]);
1380 }
1381 return AwkValue::num((double)parts.size());
1382 }
1383 if (n == "sub" || n == "gsub") {
1384 std::string pat = sval(0);
1385 std::string rep = sval(1);
1386 Expr* target = (e.args.size() >= 3) ? e.args[2].get() : nullptr;
1387 std::string subject;
1388 if (target) subject = eval(*target).asString();
1389 else subject = record;
1390 int count = 0;
1391 try {
1392 std::regex re(pat);
1393 if (n == "sub") {
1394 std::smatch m;
1395 if (std::regex_search(subject, m, re)) {
1396 subject = m.prefix().str() + std::regex_replace(m[0].str(), re, rep) + m.suffix().str();
1397 count = 1;
1398 }
1399 } else {
1400 std::string out;
1401 auto begin = std::sregex_iterator(subject.begin(), subject.end(), re);
1402 auto end = std::sregex_iterator();
1403 std::size_t prev = 0;
1404 for (auto it = begin; it != end; ++it) {
1405 out.append(subject, prev, it->position() - prev);
1406 out.append(it->format(rep));
1407 prev = it->position() + it->length();
1408 ++count;
1409 }
1410 out.append(subject, prev, std::string::npos);
1411 subject = std::move(out);
1412 }
1413 } catch (...) {}
1414 if (target) assignLValue(*target, AwkValue::str(subject));
1415 else { record = std::move(subject); splitRecord(); }
1416 return AwkValue::num((double)count);
1417 }
1418 if (n == "match") {
1419 std::string s = sval(0);
1420 std::string pat = sval(1);
1421 try {
1422 std::regex re(pat);
1423 std::smatch m;
1424 if (std::regex_search(s, m, re)) {
1425 vars["RSTART"] = AwkValue::num((double)(m.position(0) + 1));
1426 vars["RLENGTH"] = AwkValue::num((double)m.length(0));
1427 return AwkValue::num((double)(m.position(0) + 1));
1428 }
1429 } catch (...) {}
1430 vars["RSTART"] = AwkValue::num(0);
1431 vars["RLENGTH"] = AwkValue::num(-1);
1432 return AwkValue::num(0);
1433 }
1434 if (n == "tolower") {
1435 std::string s = sval(0);
1436 for (auto& c : s) c = (char)std::tolower((unsigned char)c);
1437 return AwkValue::str(s);
1438 }
1439 if (n == "toupper") {
1440 std::string s = sval(0);
1441 for (auto& c : s) c = (char)std::toupper((unsigned char)c);
1442 return AwkValue::str(s);
1443 }
1444 if (n == "sprintf" || n == "printf") {
1445 std::vector<AwkValue> a;
1446 for (auto& x : e.args) a.push_back(eval(*x));
1447 std::string out = formatPrintf(a);
1448 if (n == "sprintf") return AwkValue::str(out);
1449 std::fputs(out.c_str(), stdout);
1450 return AwkValue::str("");
1451 }
1452 if (n == "system") {
1453 std::string cmd = sval(0);
1454 int rc = std::system(cmd.c_str());
1455 return AwkValue::num((double)rc);
1456 }
1457 if (n == "int") return AwkValue::num(std::trunc(nval(0)));
1458 if (n == "sqrt") return AwkValue::num(std::sqrt(nval(0)));
1459 if (n == "exp") return AwkValue::num(std::exp(nval(0)));
1460 if (n == "log") return AwkValue::num(std::log(nval(0)));
1461 if (n == "sin") return AwkValue::num(std::sin(nval(0)));
1462 if (n == "cos") return AwkValue::num(std::cos(nval(0)));
1463 if (n == "atan2") return AwkValue::num(std::atan2(nval(0), nval(1)));
1464 if (n == "rand") return AwkValue::num((double)std::rand() / RAND_MAX);
1465 if (n == "srand") {
1466 unsigned s = (unsigned)nval(0);
1467 std::srand(s);
1468 return AwkValue::num(0);
1469 }
1470 // Unknown — return empty.
1471 return AwkValue::str("");
1472 }

1474 std::string formatPrintf(const std::vector<AwkValue>& args) {
1475 if (args.empty()) return std::string();
1476 const std::string& fmt = args[0].asString();
1477 std::string out;
1478 std::size_t ai = 1;
1479 std::size_t i = 0;
1480 while (i < fmt.size()) {
1481 char c = fmt[i];
1482 if (c != '%') {
1483 if (c == '\\' && i + 1 < fmt.size()) {
1484 char n = fmt[i + 1];
1485 switch (n) {
1486 case 'n': out.push_back('\n'); break;
1487 case 't': out.push_back('\t'); break;
1488 case 'r': out.push_back('\r'); break;
1489 case '\\': out.push_back('\\'); break;
1490 case '"': out.push_back('"'); break;
1491 default: out.push_back('\\'); out.push_back(n); break;
1492 }
1493 i += 2;
1494 continue;
1495 }
1496 out.push_back(c); ++i; continue;
1497 }
1498 // %... directive
1499 std::size_t s = i;
1500 ++i;
1501 while (i < fmt.size()
1502 && std::strchr("-+0 #", fmt[i])) ++i;
1503 while (i < fmt.size() && std::isdigit((unsigned char)fmt[i])) ++i;
1504 if (i < fmt.size() && fmt[i] == '.') {
1505 ++i;
1506 while (i < fmt.size() && std::isdigit((unsigned char)fmt[i])) ++i;
1507 }
1508 if (i >= fmt.size()) { out.append(fmt, s, std::string::npos); break; }
1509 char conv = fmt[i++];
1510 std::string spec = fmt.substr(s, i - s);
1511 char buf[256];
1512 AwkValue v = (ai < args.size()) ? args[ai++] : AwkValue::str("");
1513 if (conv == 's') {
1514 std::snprintf(buf, sizeof(buf), spec.c_str(), v.asString().c_str());
1515 out.append(buf);
1516 } else if (conv == 'c') {
1517 if (v.has_s && !v.asString().empty())
1518 std::snprintf(buf, sizeof(buf), spec.c_str(), v.asString()[0]);
1519 else
1520 std::snprintf(buf, sizeof(buf), spec.c_str(), (int)v.asNumber());
1521 out.append(buf);
1522 } else if (conv == 'd' || conv == 'i') {
1523 std::string mod = spec.substr(0, spec.size() - 1) + "ll" + conv;
1524 std::snprintf(buf, sizeof(buf), mod.c_str(), (long long)v.asNumber());
1525 out.append(buf);
1526 } else if (conv == 'o' || conv == 'x' || conv == 'X' || conv == 'u') {
1527 std::string mod = spec.substr(0, spec.size() - 1) + "ll" + conv;
1528 std::snprintf(buf, sizeof(buf), mod.c_str(), (unsigned long long)v.asNumber());
1529 out.append(buf);
1530 } else if (conv == 'f' || conv == 'e' || conv == 'E'
1531 || conv == 'g' || conv == 'G')
1532 {
1533 std::snprintf(buf, sizeof(buf), spec.c_str(), v.asNumber());
1534 out.append(buf);
1535 } else if (conv == '%') {
1536 out.push_back('%');
1537 } else {
1538 out.append(spec);
1539 }
1540 }
1541 return out;
1542 }

1544 // Run statement; throws Break/Continue/Next/Exit for control.
1545 void run(Stmt& s) {
1546 if (exiting) return;
1547 switch (s.kind) {
1548 case SK::Empty: return;
1549 case SK::Block:
1550 for (auto& c : s.children) { run(*c); if (exiting) return; }
1551 return;
1552 case SK::If: {
1553 AwkValue cond = eval(*s.exprs[0]);
1554 if (cond.truthy()) {
1555 if (!s.children.empty()) run(*s.children[0]);
1556 } else if (s.children.size() >= 2) {
1557 run(*s.children[1]);
1558 }
1559 return;
1560 }
1561 case SK::While: {
1562 while (eval(*s.exprs[0]).truthy()) {
1563 try { run(*s.children[0]); }
1564 catch (AwkBreakEx&) { return; }
1565 catch (AwkContinueEx&) { continue; }
1566 }
1567 return;
1568 }
1569 case SK::DoWhile: {
1570 do {
1571 try { run(*s.children[0]); }
1572 catch (AwkBreakEx&) { return; }
1573 catch (AwkContinueEx&) {}
1574 } while (eval(*s.exprs[0]).truthy());
1575 return;
1576 }
1577 case SK::For: {
1578 if (s.init) eval(*s.init);
1579 while (!s.cond || eval(*s.cond).truthy()) {
1580 try { run(*s.children[0]); }
1581 catch (AwkBreakEx&) { return; }
1582 catch (AwkContinueEx&) {}
1583 if (s.step) eval(*s.step);
1584 }
1585 return;
1586 }
1587 case SK::ForIn: {
1588 auto& m = arrays[s.name2];
1589 std::vector<std::string> keys;
1590 for (auto& kv : m) keys.push_back(kv.first);
1591 for (const auto& k : keys) {
1592 vars[s.name1] = AwkValue::str(k);
1593 try { run(*s.children[0]); }
1594 catch (AwkBreakEx&) { return; }
1595 catch (AwkContinueEx&) { continue; }
1596 }
1597 return;
1598 }
1599 case SK::Break: throw AwkBreakEx{};
1600 case SK::Continue: throw AwkContinueEx{};
1601 case SK::Next: throw AwkNextEx{};
1602 case SK::Exit:
1603 if (!s.exprs.empty()) exit_status = (int)eval(*s.exprs[0]).asNumber();
1604 exiting = true;
1605 throw AwkExitEx{};
1606 case SK::Delete: {
1607 Expr& tgt = *s.exprs[0];
1608 if (tgt.kind == EK::Var) { arrays.erase(tgt.name); return; }
1609 if (tgt.kind == EK::ArrayRef) {
1610 auto& m = arrays[tgt.name];
1611 m.erase(buildSubscript(tgt.args));
1612 return;
1613 }
1614 return;
1615 }
1616 case SK::Return: return; // no user functions yet
1617 case SK::Print: {
1618 std::string out;
1619 if (s.exprs.empty()) {
1620 out = record;
1621 } else {
1622 for (std::size_t i = 0; i < s.exprs.size(); ++i) {
1623 if (i) out += OFS;
1624 out += eval(*s.exprs[i]).asString();
1625 }
1626 }
1627 out += ORS;
1628 emit(out, s);
1629 return;
1630 }
1631 case SK::Printf: {
1632 std::vector<AwkValue> a;
1633 for (auto& x : s.exprs) a.push_back(eval(*x));
1634 std::string out = formatPrintf(a);
1635 emit(out, s);
1636 return;
1637 }
1638 case SK::ExprStmt:
1639 eval(*s.exprs[0]);
1640 return;
1641 }
1642 }

1645 FILE* dest = stdout;
1646 if (s.redir_kind == 1 || s.redir_kind == 2) {
1647 std::string fn = eval(*s.to_file).asString();
1648 auto it = output_files.find(fn);
1649 if (it == output_files.end()) {
1650 FILE* f = std::fopen(fn.c_str(), s.redir_kind == 1 ? "wb" : "ab");
1651 if (!f) return;
1652 output_files[fn] = f;
1653 dest = f;
1654 } else dest = it->second;
1655 } else if (s.redir_kind == 3) {
1656 std::string cmd = eval(*s.to_file).asString();
1657#ifdef _WIN32

1659#else

1670 }
1671 slot = f;

1673 }
1674 std::fwrite(out.data(), 1, out.size(), dest);
1675 }

1726 void runAwkRule(AwkRule& r) {
1727 bool match = false;
1728 switch (r.pat.kind) {
1729 case AwkPattern::Always: match = true; break;
1730 case AwkPattern::Expr: match = evalAwkPatternExpr(*r.pat.e1); break;
1731 case AwkPattern::Range:
1732 if (!r.in_range) {
1733 if (evalAwkPatternExpr(*r.pat.e1)) { r.in_range = true; match = true; }
1734 } else {
1735 match = true;
1736 if (evalAwkPatternExpr(*r.pat.e2)) r.in_range = false;
1737 }
1738 break;
1739 default: return; // BEGIN/END handled separately
1740 }
1741 if (match) try { run(*r.action); }
1742 catch (AwkNextEx&) { throw; }
1743 catch (AwkExitEx&) { throw; }
1744 }

1751 catch (...) {}

1762 void processStream(FILE* f, const std::string& fname) {
1763 FILENAME = fname;
1764 FNR = 0;
1765 std::string line;
1766 int c;
1767 while (!exiting) {
1768 line.clear();
1769 bool any = false;
1770 while ((c = std::fgetc(f)) != EOF) {
1771 any = true;
1772 if (c == '\n') break;
1773 line.push_back((char)c);
1774 }
1775 if (!any) break;
1776 record = std::move(line);
1777 splitRecord();
1778 ++NR; ++FNR;
1779 try {
1780 for (auto& r : prog.rules) {
1781 if (r.pat.kind == AwkPattern::Begin || r.pat.kind == AwkPattern::End) continue;
1782 runAwkRule(r);
1783 }
1784 } catch (AwkNextEx&) { continue; }
1785 catch (AwkExitEx&) { return; }
1786 }

1790int builtin_awk(Executor& exec, const std::vector<std::string>& args) {
1791 std::string program_text;
1792 std::string field_sep;
1793 std::vector<std::string> files;
1794 std::vector<std::pair<std::string, std::string>> var_assigns;
1795 bool have_program = false;
1796 for (std::size_t i = 0; i < args.size(); ++i) {
1797 const std::string& a = args[i];
1798 if (a == "-F" && i + 1 < args.size()) { field_sep = args[++i]; continue; }
1799 if (a.size() > 2 && a.compare(0, 2, "-F") == 0) { field_sep = a.substr(2); continue; }
1800 if (a == "-f" && i + 1 < args.size()) {
1801 std::string p = exec.pathConv().toWin32(args[++i]);
1802 std::ifstream f(p, std::ios::binary);
1803 if (!f) {
1804 std::fprintf(stderr, "awk: cannot open program file: %s\n", args[i].c_str());
1805 return 2;
1806 }
1807 std::ostringstream ss; ss << f.rdbuf();
1808 program_text = ss.str();
1809 have_program = true;
1810 continue;
1811 }
1812 if (a == "-v" && i + 1 < args.size()) {
1813 std::string kv = args[++i];
1814 auto eq = kv.find('=');
1815 if (eq != std::string::npos) {
1816 var_assigns.emplace_back(kv.substr(0, eq), kv.substr(eq + 1));
1817 }
1818 continue;
1819 }
1820 if (a == "--") {
1821 ++i;
1822 if (!have_program && i < args.size()) {
1823 program_text = args[i++];
1824 have_program = true;
1825 }
1826 while (i < args.size()) files.push_back(args[i++]);
1827 break;
1828 }
1829 if (!have_program) {
1830 program_text = a;
1831 have_program = true;
1832 continue;
1833 }
1834 files.push_back(a);
1835 }
1836 if (!have_program) {
1837 std::fprintf(stderr, "awk: missing program text\n");
1838 return 2;
1839 }
1840 AwkProgram prog;
1841 try {
1842 AwkParser parser(program_text);
1843 prog = parser.parseAwkProgram();
1844 } catch (const std::exception& ex) {
1845 std::fprintf(stderr, "%s\n", ex.what());
