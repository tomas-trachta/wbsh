18 static void escapeString(std::ostream& os, const std::string& s) {
19 for (char c : s) {
20 switch (c) {
21 case '\n': os << "\\n"; break;
22 case '\t': os << "\\t"; break;
23 case '\r': os << "\\r"; break;
24 case '\\': os << "\\\\"; break;
25 case '"': os << "\\\""; break;
26 default: os << c;
27 }
28 }

87 static void dumpNode(std::ostream& os, int depth, const Node& n) {
88 indent(os, depth);
89 os << nodeKindName(n.kind) << "\n";
90 switch (n.kind) {
91 case Node::Kind::SimpleCommand: {
92 const auto& sc = static_cast<const SimpleCommand&>(n);
93 for (const auto& a : sc.assignments) {
94 indent(os, depth + 1);
95 os << "Assign " << a.name << "=";
96 dumpSegments(os, a.value.segments);
97 os << "\n";
98 }
99 for (const auto& w : sc.words) dumpWord(os, depth + 1, "Word", w);
100 for (const auto& r : sc.redirs) dumpRedir(os, depth + 1, r);
101 break;
102 }
103 case Node::Kind::Pipeline: {
104 const auto& p = static_cast<const Pipeline&>(n);
105 if (p.bang) { indent(os, depth + 1); os << "(bang)\n"; }
106 for (std::size_t i = 0; i < p.commands.size(); ++i) {
107 if (i > 0) {
108 indent(os, depth + 1);
109 os << (p.stderr_to_stdout[i - 1] ? "|& pipe\n" : "| pipe\n");
110 }
111 dumpNode(os, depth + 1, *p.commands[i]);
112 }
113 break;
114 }
115 case Node::Kind::AndOr: {
116 const auto& ao = static_cast<const AndOr&>(n);
117 indent(os, depth + 1);
118 os << (ao.op == AndOr::Op::AndIf ? "&&" : "||") << "\n";
119 dumpNode(os, depth + 1, *ao.left);
120 dumpNode(os, depth + 1, *ao.right);
121 break;
122 }
123 case Node::Kind::List: {
124 const auto& l = static_cast<const List&>(n);
125 for (const auto& it : l.items) {
126 indent(os, depth + 1);
127 os << "Item" << (it.background ? " (bg)" : "") << "\n";
128 dumpNode(os, depth + 2, *it.command);
129 }
130 break;
131 }
132 case Node::Kind::BraceGroup: {
133 const auto& bg = static_cast<const BraceGroup&>(n);
134 dumpListBody(os, depth + 1, bg.body.get());
135 for (const auto& r : bg.redirs) dumpRedir(os, depth + 1, r);
136 break;
137 }
138 case Node::Kind::Subshell: {
139 const auto& ss = static_cast<const Subshell&>(n);
140 dumpListBody(os, depth + 1, ss.body.get());
141 for (const auto& r : ss.redirs) dumpRedir(os, depth + 1, r);
142 break;
143 }
144 case Node::Kind::IfClause: {
145 const auto& ic = static_cast<const IfClause&>(n);
146 for (std::size_t i = 0; i < ic.branches.size(); ++i) {
147 indent(os, depth + 1);
148 os << (i == 0 ? "if-cond\n" : "elif-cond\n");
149 dumpListBody(os, depth + 2, ic.branches[i].cond.get());
150 indent(os, depth + 1);
151 os << "then\n";
152 dumpListBody(os, depth + 2, ic.branches[i].body.get());
153 }
154 if (ic.else_body) {
155 indent(os, depth + 1); os << "else\n";
156 dumpListBody(os, depth + 2, ic.else_body.get());
157 }
158 for (const auto& r : ic.redirs) dumpRedir(os, depth + 1, r);
159 break;
160 }
161 case Node::Kind::WhileClause: {
162 const auto& w = static_cast<const WhileClause&>(n);
163 indent(os, depth + 1);
164 os << (w.until ? "until-cond\n" : "while-cond\n");
165 dumpListBody(os, depth + 2, w.cond.get());
166 indent(os, depth + 1); os << "do\n";
167 dumpListBody(os, depth + 2, w.body.get());
168 for (const auto& r : w.redirs) dumpRedir(os, depth + 1, r);
169 break;
170 }
171 case Node::Kind::ForClause: {
172 const auto& f = static_cast<const ForClause&>(n);
173 indent(os, depth + 1);
174 os << "var " << f.var << "\n";
175 if (f.has_in) {
176 indent(os, depth + 1); os << "in\n";
177 for (const auto& w : f.items) dumpWord(os, depth + 2, "Word", w);
178 }
179 indent(os, depth + 1); os << "do\n";
180 dumpListBody(os, depth + 2, f.body.get());
181 for (const auto& r : f.redirs) dumpRedir(os, depth + 1, r);
182 break;
183 }
184 case Node::Kind::CaseClause: {
185 const auto& c = static_cast<const CaseClause&>(n);
186 dumpWord(os, depth + 1, "subject", c.subject);
187 for (const auto& it : c.items) {
188 indent(os, depth + 1);
189 os << "Case-Item term=";
190 switch (it.term) {
191 case CaseClause::Term::DSemi: os << ";;"; break;
192 case CaseClause::Term::SemiAmp: os << ";&"; break;
193 case CaseClause::Term::DSemiAmp: os << ";;&"; break;
194 }
195 os << "\n";
196 indent(os, depth + 2); os << "patterns:\n";
197 for (const auto& p : it.patterns) dumpWord(os, depth + 3, "Word", p);
198 indent(os, depth + 2); os << "body:\n";
199 dumpListBody(os, depth + 3, it.body.get());
200 }
201 for (const auto& r : c.redirs) dumpRedir(os, depth + 1, r);
202 break;
203 }
204 case Node::Kind::FunctionDef: {
205 const auto& f = static_cast<const FunctionDef&>(n);
206 indent(os, depth + 1); os << "name " << f.name << "\n";
207 indent(os, depth + 1); os << "body:\n";
208 dumpNode(os, depth + 2, *f.body);
209 break;
210 }
211 case Node::Kind::DBracket: {
212 const auto& d = static_cast<const DBracketCond&>(n);
213 if (d.root) {
214 indent(os, depth + 1); os << "expr:\n";
215 std::function<void(const DBracketCond::Expr&, int)> dump =
216 [&](const DBracketCond::Expr& e, int dd) {
217 using K = DBracketCond::Expr::K;
218 indent(os, dd);
219 switch (e.k) {
220 case K::And: os << "&&\n"; dump(*e.a, dd + 1); dump(*e.b, dd + 1); break;
221 case K::Or: os << "||\n"; dump(*e.a, dd + 1); dump(*e.b, dd + 1); break;
222 case K::Not: os << "!\n"; dump(*e.a, dd + 1); break;
223 case K::Prim:
224 if (e.op.empty()) os << "test \"" << e.lhs.raw << "\"\n";
225 else if (e.rhs.raw.empty())
226 os << "test " << e.op << " \"" << e.lhs.raw << "\"\n";
227 else
228 os << "test \"" << e.lhs.raw << "\" "
229 << e.op << " \"" << e.rhs.raw << "\"\n";
230 break;
231 }
232 };
233 dump(*d.root, depth + 2);
234 }
235 for (const auto& r : d.redirs) dumpRedir(os, depth + 1, r);
236 break;
237 }
238 }

241 void dumpTokens(std::ostream& os, const std::vector<Token>& tokens) {
242 for (const auto& t : tokens) {
243 os << t.loc.line << ":" << t.loc.column << " "
244 << tokKindName(t.kind);
245 if (t.kind == TokKind::Word || t.kind == TokKind::IoNumber) {
246 os << " text=\"";
247 escapeString(os, t.text);
248 os << "\" segs=";
249 dumpSegments(os, t.segments);
250 }
251 if (t.is_heredoc_delim) {
252 os << " (heredoc-delim, quoted=" << (t.heredoc_quoted ? "y" : "n")
253 << ", strip_tabs=" << (t.heredoc_strip_tabs ? "y" : "n")
254 << ", body=\"";
255 escapeString(os, t.heredoc_body);
256 os << "\")";
257 }
258 os << "\n";
259 }
