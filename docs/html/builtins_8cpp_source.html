39 static long long toIntSafe(const std::string& s, bool& ok) {
40 ok = false;
41 if (s.empty()) return 0;
42 try {
43 std::size_t idx = 0;
44 long long v = std::stoll(s, &idx, 10);
45 if (idx != s.size()) return 0;
46 ok = true;
47 return v;
48 } catch (...) { return 0; }
49 }

51 // Parse args[0] as a small int. Returns `fallback` when args is empty,
52 // the value isn't a parseable integer, or (when require_positive)
53 // it's <= 0. Centralises the "first arg is an optional count" idiom
54 // used by exit/return/break/continue/shift.
55 static int firstArgAsInt(const std::vector<std::string>& args, int fallback,
56 bool require_positive = false) {
57 if (args.empty()) return fallback;
58 bool ok = false;
59 long long v = toIntSafe(args[0], ok);
60 if (!ok) return fallback;
61 if (require_positive && v <= 0) return fallback;
62 return static_cast<int>(v);
63 }

77 static std::string interpretEcho(const std::string& s) {
78 std::string out;
79 for (std::size_t i = 0; i < s.size(); ++i) {
80 char c = s[i];
81 if (c != '\\' || i + 1 >= s.size()) { out.push_back(c); continue; }
82 char nx = s[++i];
83 switch (nx) {
84 case 'a': out.push_back('\a'); break;
85 case 'b': out.push_back('\b'); break;
86 case 'e': out.push_back('\x1b'); break;
87 case 'f': out.push_back('\f'); break;
88 case 'n': out.push_back('\n'); break;
89 case 'r': out.push_back('\r'); break;
90 case 't': out.push_back('\t'); break;
91 case 'v': out.push_back('\v'); break;
92 case '\\': out.push_back('\\'); break;
93 case '0': {
94 int val = 0, cnt = 0;
95 while (cnt < 3 && i + 1 < s.size() && s[i + 1] >= '0' && s[i + 1] <= '7') {
96 val = val * 8 + (s[++i] - '0');
97 ++cnt;
98 }
99 out.push_back(static_cast<char>(val));
100 break;
101 }
102 case 'x': {
103 int val = 0, cnt = 0;
104 while (cnt < 2 && i + 1 < s.size() && std::isxdigit(static_cast<unsigned char>(s[i + 1]))) {
105 char h = s[++i];
106 int d = std::isdigit(static_cast<unsigned char>(h))
107 ? h - '0'
108 : std::tolower(static_cast<unsigned char>(h)) - 'a' + 10;
109 val = val * 16 + d;
110 ++cnt;
111 }
112 out.push_back(static_cast<char>(val));
113 break;
114 }
115 default: out.push_back('\\'); out.push_back(nx); break;
116 }
117 }
118 return out;
119 }

121 static int builtin_echo(Executor&, const std::vector<std::string>& args) {
122 bool newline = true;
123 bool interp = false;
124 std::size_t i = 0;
125 while (i < args.size() && !args[i].empty() && args[i][0] == '-') {
126 const std::string& f = args[i];
127 if (f == "--") { ++i; break; }
128 bool ok = f.size() > 1;
129 for (std::size_t k = 1; k < f.size(); ++k) {
130 if (f[k] == 'n') newline = false;
131 else if (f[k] == 'e') interp = true;
132 else if (f[k] == 'E') interp = false;
133 else { ok = false; break; }
134 }
135 if (!ok) break;
136 ++i;
137 }
138 bool first = true;
139 for (; i < args.size(); ++i) {
140 if (!first) std::fputc(' ', stdout);
141 first = false;
142 if (interp) {
143 std::string s = interpretEcho(args[i]);
144 std::fwrite(s.data(), 1, s.size(), stdout);
145 } else {
146 std::fwrite(args[i].data(), 1, args[i].size(), stdout);
147 }
148 }
149 if (newline) std::fputc('\n', stdout);
150 std::fflush(stdout);
151 return 0;
152 }

156 static int builtin_printf(Executor&, const std::vector<std::string>& args) {
157 if (args.empty()) {
158 printerr("printf: missing format");
159 return 2;
160 }
161 const std::string& fmt = args[0];
162 std::size_t ai = 1;
163 auto arg = [&](bool& consumed) -> std::string {
164 consumed = true;
165 if (ai < args.size()) return args[ai++];
166 consumed = false;
167 return {};
168 };
169 auto emit_once = [&]() {
170 for (std::size_t i = 0; i < fmt.size(); ++i) {
171 char c = fmt[i];
172 if (c == '\\' && i + 1 < fmt.size()) {
173 char nx = fmt[++i];
174 switch (nx) {
175 case 'a': std::fputc('\a', stdout); break;
176 case 'b': std::fputc('\b', stdout); break;
177 case 'e': std::fputc('\x1b', stdout); break;
178 case 'f': std::fputc('\f', stdout); break;
179 case 'n': std::fputc('\n', stdout); break;
180 case 'r': std::fputc('\r', stdout); break;
181 case 't': std::fputc('\t', stdout); break;
182 case 'v': std::fputc('\v', stdout); break;
183 case '\\': std::fputc('\\', stdout); break;
184 default: std::fputc('\\', stdout); std::fputc(nx, stdout); break;
185 }
186 continue;
187 }
188 if (c != '%') { std::fputc(c, stdout); continue; }
189 // Parse conversion spec.
190 std::size_t spec_start = i;
191 ++i;
192 std::string spec = "%";
193 while (i < fmt.size() && std::strchr("-+ #0", fmt[i])) spec.push_back(fmt[i++]);
194 while (i < fmt.size() && std::isdigit(static_cast<unsigned char>(fmt[i]))) spec.push_back(fmt[i++]);
195 if (i < fmt.size() && fmt[i] == '.') {
196 spec.push_back(fmt[i++]);
197 while (i < fmt.size() && std::isdigit(static_cast<unsigned char>(fmt[i]))) spec.push_back(fmt[i++]);
198 }
199 if (i >= fmt.size()) {
200 std::fwrite(fmt.data() + spec_start, 1, fmt.size() - spec_start, stdout);
201 break;
202 }
203 char conv = fmt[i];
204 spec.push_back(conv);
205 bool consumed;
206 std::string a = arg(consumed);
207 char buf[64];
208 switch (conv) {
209 case 's': std::fprintf(stdout, spec.c_str(), a.c_str()); break;
210 case 'd': case 'i': {
211 bool ok; long long v = toIntSafe(a, ok);
212 std::string s2 = spec; s2.pop_back(); s2 += "lld";
213 std::fprintf(stdout, s2.c_str(), v);
214 break;
215 }
216 case 'u': {
217 bool ok; long long v = toIntSafe(a, ok);
218 unsigned long long uv = (unsigned long long)v;
219 std::string s2 = spec; s2.pop_back(); s2 += "llu";
220 std::fprintf(stdout, s2.c_str(), uv);
221 break;
222 }
223 case 'x': case 'X': case 'o': {
224 bool ok; long long v = toIntSafe(a, ok);
225 unsigned long long uv = (unsigned long long)v;
226 std::string s2 = spec; s2.pop_back(); s2 += "ll";
227 s2.push_back(conv);
228 std::fprintf(stdout, s2.c_str(), uv);
229 break;
230 }
231 case 'c': {
232 char ch = a.empty() ? '\0' : a[0];
233 std::fputc(ch, stdout);
234 (void)buf;
235 break;
236 }
237 case '%': std::fputc('%', stdout); break;
238 default:
239 std::fwrite(spec.data(), 1, spec.size(), stdout);
240 break;
241 }
242 }
243 };
244 emit_once();
245 while (ai < args.size()) {
246 std::size_t before = ai;
247 emit_once();
248 if (ai == before) break;
249 }
250 std::fflush(stdout);
251 return 0;
252 }

431 static int builtin_exec(Executor& exec, const std::vector<std::string>& args) {
432 // Bash semantics: with no command, redirections take effect on
433 // the current shell (permanent for the rest of the session).
434 // With a command, replaces the shell with that command — we
435 // can't do the latter losslessly on Windows so we just run it
436 // and exit with its status.
437 if (args.empty()) return 0; // redirections-only handled by caller
438 // Run as if it were a regular external invocation.
439 std::vector<std::string> argv = args;
440 std::string cmd = argv[0];
441 argv.erase(argv.begin());
442 if (exec.isBuiltin(cmd)) return exec.callBuiltin(cmd, argv);
443 if (exec.isFunction(cmd)) return exec.callFunction(cmd, argv);
444 std::string joined;
445 for (std::size_t i = 0; i < args.size(); ++i) {
446 if (i) joined.push_back(' ');
447 joined.push_back('\'');
448 for (char c : args[i]) {
449 if (c == '\'') joined += "'\\''";
450 else joined.push_back(c);
451 }
452 joined.push_back('\'');
453 }
454 int rc = exec.executeText(joined, "<exec>");
455 throw ShellExit{ rc };
456 }

527 if (!found.empty()) break;
528#endif
529 }
530 if (!found.empty()) {

542 std::size_t i = 0;
543 bool show_path = false;
544 bool show_verbose = false;
545 while (i < args.size() && !args[i].empty() && args[i][0] == '-') {
546 const std::string& f = args[i];
547 if (f == "--") { ++i; break; }
548 if (f == "-v") { show_path = true; ++i; continue; }
549 if (f == "-V") { show_verbose = true; ++i; continue; }
550 break;
551 }
552 if (i >= args.size()) return 0;

555 return builtin_type(exec, rest);

567 if (k) joined.push_back(' ');

569 joined.push_back('"');
570 for (char c : rest[k]) {
571 if (c == '"' || c == '\\' || c == '$' || c == '`') joined.push_back('\\');
572 joined.push_back(c);
573 }
574 joined.push_back('"');
575 }
576 return exec.executeText(joined, "command");

586 const std::string& f = args[i];

588 if (f == "-r") { raw = true; ++i; continue; }
589 if (f == "-p") {

592 }

609 if (n == EOF) break;
610 if (n == '\n') continue; // line continuation
611 line.push_back(static_cast<char>(n));

615 }

618 if (names.empty()) names.push_back("REPLY");

620 std::string ifs = exec.env().get("IFS");
621 if (ifs.empty()) ifs = " \t\n";
622 auto isIfsWS = [&](char c) {
623 return (c == ' ' || c == '\t' || c == '\n') && ifs.find(c) != std::string::npos;
624 };
625 auto isIfs = [&](char c) { return ifs.find(c) != std::string::npos; };
626
627 std::vector<std::string> fields;
628 std::size_t pos = 0;
629 while (pos < line.size() && isIfsWS(line[pos])) ++pos;

663 std::string path = pc.toWin32(raw_path);

669 case 'd': return (ok && (st.st_mode & S_IFMT) == S_IFDIR) ? 0 : 1;
670 case 's': return (ok && st.st_size > 0) ? 0 : 1;
671 case 'r': return (ok && (st.st_mode & 0444)) ? 0 : 1;
672 case 'w': return (ok && (st.st_mode & 0222)) ? 0 : 1;
673 case 'x': return (ok && (st.st_mode & 0111)) ? 0 : 1;

678 static int evalTest(const std::vector<std::string>& a, const PathConv& pc) {
679 if (a.empty()) return 1;
680
681 // Leading ! negates the rest.
682 if (a[0] == "!" && a.size() > 1) {
683 std::vector<std::string> rest(a.begin() + 1, a.end());
684 int r = evalTest(rest, pc);
685 if (r == 2) return 2;
686 return r == 0 ? 1 : 0;
687 }

689 auto evalOne = [&](const std::string& s) {
690 return s.empty() ? 1 : 0;
691 };
692 auto strNum = [&](const std::string& s, bool& ok) -> long long {
693 return toIntSafe(s, ok);
694 };
695
696 if (a.size() == 1) return evalOne(a[0]);
697
698 if (a.size() == 2) {
699 if (a[0].size() == 2 && a[0][0] == '-') {
700 char op = a[0][1];
701 if (op == 'z') return a[1].empty() ? 0 : 1;
702 if (op == 'n') return a[1].empty() ? 1 : 0;
703 return evalUnaryFileTest(op, a[1], pc);
704 }
705 return 2;
706 }
707
708 if (a.size() == 3) {
709 const std::string& l = a[0]; const std::string& op = a[1]; const std::string& r = a[2];
710 if (op == "=" || op == "==") return l == r ? 0 : 1;
711 if (op == "!=") return l != r ? 0 : 1;
712 if (op == "<") return l < r ? 0 : 1;
713 if (op == ">") return l > r ? 0 : 1;
714 bool okL, okR;
715 long long li = strNum(l, okL); long long ri = strNum(r, okR);
716 if (okL && okR) {
717 if (op == "-eq") return li == ri ? 0 : 1;
718 if (op == "-ne") return li != ri ? 0 : 1;
719 if (op == "-lt") return li < ri ? 0 : 1;
720 if (op == "-le") return li <= ri ? 0 : 1;
721 if (op == "-gt") return li > ri ? 0 : 1;
722 if (op == "-ge") return li >= ri ? 0 : 1;
723 }
724 return 2;
725 }
726 return 2;
727 }

860 static int builtin_shopt(Executor& exec, const std::vector<std::string>& args) {
861 enum class Mode { ListSet, ListUnset, ListAll, Set, Unset, Query };
862 Mode mode = Mode::ListAll;
863 bool printable = false;
864 std::vector<std::string> names;
865 for (const auto& a : args) {
866 if (a == "-s") mode = Mode::Set;
867 else if (a == "-u") mode = Mode::Unset;
868 else if (a == "-q") mode = Mode::Query;
869 else if (a == "-p") printable = true;
870 else if (!a.empty() && a[0] == '-' && a != "--") {
871 std::fprintf(stderr, "wbsh: shopt: unknown option: %s\n", a.c_str());

873 }
874 else if (a == "--") { /* end of opts */ }

876 }
877 auto findFlag = [](const std::string& n) -> const ShoptFlag* {
878 for (const ShoptFlag* p = shoptTable(); p->name; ++p) {
879 if (n == p->name) return p;
880 }
881 return nullptr;
882 };
883 (void)printable;
884 if (mode == Mode::Set || mode == Mode::Unset) {
885 bool on = (mode == Mode::Set);
886 int rc = 0;
887 for (const auto& n : names) {
888 auto* f = findFlag(n);
889 if (!f) {
890 std::fprintf(stderr, "wbsh: shopt: %s: invalid option name\n", n.c_str());
891 rc = 1;
892 continue;

894 (exec.env().*(f->set))(on);
895 }
896 return rc;
897 }
898 if (mode == Mode::Query) {
899 for (const auto& n : names) {
900 auto* f = findFlag(n);
901 if (!f) return 1;
902 if (!(exec.env().*(f->get))()) return 1;
903 }
904 return 0;
905 }
906 // List form. With names: print only those. Without names: all.
907 auto print_one = [&](const ShoptFlag* f) {
908 bool on = (exec.env().*(f->get))();

912 int rc = 0;
913 for (const auto& n : names) {
914 auto* f = findFlag(n);
915 if (!f) {
916 std::fprintf(stderr, "wbsh: shopt: %s: invalid option name\n", n.c_str());
917 rc = 1;
918 continue;
919 }
920 print_one(f);
921 }
922 return rc;
923 }
924 for (const ShoptFlag* p = shoptTable(); p->name; ++p) print_one(p);
925 return 0;
926 }

928 static int builtin_mapfile(Executor& exec, const std::vector<std::string>& args) {
929 bool strip_newline = false;
930 long long max_lines = -1; // -n N: stop after N lines
931 long long origin = 0; // -O start: assign starting from index
932 long long skip = 0; // -s N: skip first N lines from input
933 std::string array_name = "MAPFILE";
934 for (std::size_t i = 0; i < args.size(); ++i) {
935 const std::string& a = args[i];
936 if (a == "-t") strip_newline = true;
937 else if (a == "-n" && i + 1 < args.size()) {
938 try { max_lines = std::stoll(args[++i]); } catch (...) {}
939 }
940 else if (a == "-O" && i + 1 < args.size()) {
941 try { origin = std::stoll(args[++i]); } catch (...) {}
942 }
943 else if (a == "-s" && i + 1 < args.size()) {
944 try { skip = std::stoll(args[++i]); } catch (...) {}
945 }
946 else if (a == "-u" && i + 1 < args.size()) {
947 // File descriptor — we only support 0 (stdin) here.
948 try { (void)std::stoi(args[++i]); } catch (...) {}
949 }
950 else if (!a.empty() && a[0] == '-' && a != "-" && a != "--") {
951 std::fprintf(stderr, "wbsh: mapfile: unknown option: %s\n", a.c_str());
952 return 1;
953 }
954 else array_name = a;
955 }
956 std::vector<std::string> lines;
957 std::string buf;
958 int c;
959 long long skipped = 0;
960 while ((c = std::fgetc(stdin)) != EOF) {
961 if (c == '\n') {
962 if (skipped < skip) { ++skipped; buf.clear(); continue; }
963 if (!strip_newline) buf.push_back('\n');
964 lines.push_back(std::move(buf));
965 buf.clear();
966 if (max_lines > 0 && (long long)lines.size() >= max_lines) break;
967 } else {
968 buf.push_back((char)c);
969 }
970 }
971 if (!buf.empty()) {
972 if (skipped < skip) { /* skip last partial too */ }
973 else {
974 if (max_lines <= 0 || (long long)lines.size() < max_lines)
975 lines.push_back(std::move(buf));
976 }
977 }
978 std::map<long long, std::string> elems;
979 for (std::size_t i = 0; i < lines.size(); ++i) {
980 elems[origin + (long long)i] = std::move(lines[i]);
981 }
982 exec.env().setIndexedArraySparse(array_name, std::move(elems));
983 return 0;
984 }

1326 // Append all PATH-resolvable executables to `out`. Paths are walked
1327 // in order; duplicates dropped.
1328 static void collectCommandsFromPath(Executor& exec, std::vector<std::string>& out) {
1329 std::string path = exec.env().get("PATH");
1330 if (path.empty()) return;
1331 std::set<std::string> seen;
1332 std::size_t i = 0;
1333 while (i <= path.size()) {
1334 auto sep = path.find(':', i);
1335 std::string dir = path.substr(i, sep == std::string::npos
1336 ? std::string::npos : sep - i);
1337 i = (sep == std::string::npos) ? path.size() + 1 : sep + 1;
1338 if (dir.empty()) continue;
1339 std::filesystem::path win = utf8ToPath(exec.pathConv().toWin32(dir));
1340 std::error_code ec;
1341 std::filesystem::directory_iterator it(win, ec);
1342 if (ec) continue;
1343 for (auto& e : it) {
1344 std::string n;
1345 try { n = pathToUtf8(e.path().filename()); }
1346 catch (...) { continue; }
1347 // Strip .exe / .cmd / .bat for nicer suggestions.
1348 auto ends_with = [&](const std::string& suf) {
1349 if (n.size() < suf.size()) return false;
1350 std::string tail = n.substr(n.size() - suf.size());
1351 for (auto& c : tail) c = (char)std::tolower((unsigned char)c);
1352 return tail == suf;
1353 };
1354 std::string base = n;
1355 for (const char* s : { ".exe", ".cmd", ".bat", ".com" }) {
1356 if (ends_with(s)) {
1357 base = n.substr(0, n.size() - std::strlen(s));
1358 break;
1359 }
1360 }
1361 if (seen.insert(base).second) out.push_back(base);
1362 }
1363 }
1364 }

1367 static std::vector<std::string> filterByPrefix(const std::vector<std::string>& v,
1368 const std::string& prefix) {

1370 for (const auto& s : v) {
1371 if (s.compare(0, prefix.size(), prefix) == 0)
1372 out.push_back(s);
1373 }
1374 return out;

1377 // Shared candidate generator used by compgen and (later) the line
1378 // editor's programmable-completion path. Reads -A action / -W words /
1379 // -f / -d / -c flags from `args` and produces the prefix-filtered
1380 // candidate list.
1381 static std::vector<std::string> generateCompletions(
1382 Executor& exec,
1383 const std::vector<std::string>& args,
1384 /*out*/ std::string& prefix_out)
1385 {
1386 std::vector<std::string> out;
1387 std::string prefix;
1388 std::vector<std::string> wordlist;
1389 std::string action;
1390 bool include_files = false;
1391 bool include_dirs = false;
1392 bool include_cmds = false;
1393 bool include_builtins = false;
1394 bool include_funcs = false;
1395 bool include_aliases = false;
1396 bool include_vars = false;
1397 bool include_keywords = false;
1398 for (std::size_t i = 0; i < args.size(); ++i) {
1399 const std::string& a = args[i];
1400 if (a == "-W" && i + 1 < args.size()) {
1401 std::string s = args[++i];
1402 std::string cur;
1403 for (char c : s) {
1404 if (c == ' ' || c == '\t' || c == '\n') {
1405 if (!cur.empty()) { wordlist.push_back(std::move(cur)); cur.clear(); }
1406 } else cur.push_back(c);
1407 }
1408 if (!cur.empty()) wordlist.push_back(std::move(cur));
1409 }
1410 else if (a == "-A" && i + 1 < args.size()) action = args[++i];
1411 else if (a == "-f") include_files = true;
1412 else if (a == "-d") include_dirs = true;
1413 else if (a == "-c") include_cmds = true;
1414 else if (a == "-b") include_builtins = true;
1415 else if (a == "-a") include_aliases = true;
1416 else if (a == "-v") include_vars = true;
1417 else if (a == "-k") include_keywords = true;
1418 else if (a == "-u" || a == "-g" || a == "-s" || a == "-e") {
1419 /* user/group/service/exported — skip on Windows */
1420 }
1421 else if (a == "-o" && i + 1 < args.size()) ++i; // ignore option
1422 else if (a == "-F" && i + 1 < args.size()) ++i; // function: not invoked here
1423 else if (a == "-C" && i + 1 < args.size()) ++i; // external cmd: not invoked here
1424 else if (!a.empty() && a[0] == '-' && a != "-" && a != "--") {
1425 // unknown option — ignore
1426 }
1427 else if (a == "--") { /* end of options */ }
1428 else prefix = a;
1429 }
1430 if (action == "function") include_funcs = true;
1431 if (action == "variable") include_vars = true;
1432 if (action == "alias") include_aliases = true;
1433 if (action == "builtin") include_builtins = true;
1434 if (action == "command") include_cmds = true;
1435 if (action == "file") include_files = true;
1436 if (action == "directory") include_dirs = true;
1437
1438 if (!wordlist.empty()) {
1439 for (auto& w : filterByPrefix(wordlist, prefix)) out.push_back(w);
1440 }
1441 if (include_funcs) {
1442 auto names = exec.functionNames();
1443 std::sort(names.begin(), names.end());
1444 for (auto& n : filterByPrefix(names, prefix)) out.push_back(n);
1445 }
1446 if (include_builtins) {
1447 auto names = exec.builtinNames();
1448 std::sort(names.begin(), names.end());
1449 for (auto& n : filterByPrefix(names, prefix)) out.push_back(n);
1450 }
1451 if (include_aliases) {
1452 std::vector<std::string> names;
1453 for (const auto& kv : exec.aliases()) names.push_back(kv.first);
1454 std::sort(names.begin(), names.end());
1455 for (auto& n : filterByPrefix(names, prefix)) out.push_back(n);
1456 }
1457 if (include_vars) {
1458 std::vector<std::string> names;
1459 for (const auto& kv : exec.env().vars()) names.push_back(kv.first);
1460 std::sort(names.begin(), names.end());
1461 for (auto& n : filterByPrefix(names, prefix)) out.push_back(n);
1462 }
1463 if (include_cmds) {
1464 auto names = exec.builtinNames();
1465 auto fns = exec.functionNames();
1466 names.insert(names.end(), fns.begin(), fns.end());
1467 collectCommandsFromPath(exec, names);
1468 std::sort(names.begin(), names.end());
1469 names.erase(std::unique(names.begin(), names.end()), names.end());
1470 for (auto& n : filterByPrefix(names, prefix)) out.push_back(n);
1471 }
1472 if (include_keywords) {
1473 static const char* kw[] = {
1474 "if","then","else","elif","fi","case","esac","for",
1475 "while","until","do","done","function","in","select",
1476 "time","[[","]]","return","break","continue", nullptr };
1477 std::vector<std::string> v;
1478 for (int k = 0; kw[k]; ++k) v.push_back(kw[k]);
1479 for (auto& n : filterByPrefix(v, prefix)) out.push_back(n);
1480 }
1481 if (include_files || include_dirs) {
1482 namespace fs = std::filesystem;
1483 // Split prefix into dir + leaf.
1484 std::string dir = ".", leaf = prefix;
1485 auto sl = prefix.find_last_of('/');
1486 if (sl != std::string::npos) {
1487 dir = prefix.substr(0, sl);
1488 leaf = prefix.substr(sl + 1);
1489 if (dir.empty()) dir = "/";
1490 }
1491 std::error_code ec;
1492 fs::path list_dir = utf8ToPath(exec.pathConv().toWin32(dir));
1493 fs::directory_iterator it(list_dir, ec);
1494 if (!ec) {
1495 for (auto& e : it) {
1496 std::string n;
1497 try { n = pathToUtf8(e.path().filename()); }
1498 catch (...) { continue; }
1499 if (n.empty() || n[0] == '.') continue;
1500 if (n.compare(0, leaf.size(), leaf) != 0) continue;
1501 bool isdir = e.is_directory(ec);
1502 if (include_dirs && !isdir && !include_files) continue;
1503 std::string full = (dir == ".") ? n
1504 : (dir == "/" ? "/" + n : dir + "/" + n);
1505 if (isdir) full.push_back('/');
1506 out.push_back(full);
1507 }
1508 }
1509 }
1510 prefix_out = prefix;
1511 return out;
1512 }

1521 static int builtin_complete(Executor& exec, const std::vector<std::string>& args) {
1522 Executor::CompletionSpec spec;
1523 std::vector<std::string> commands;
1524 bool remove_mode = false;
1525 bool print_mode = false;
1526 bool default_complete = false;
1527 (void)default_complete;
1528 for (std::size_t i = 0; i < args.size(); ++i) {
1529 const std::string& a = args[i];
1530 if (a == "-r") remove_mode = true;
1531 else if (a == "-p") print_mode = true;
1532 else if (a == "-D") default_complete = true;
1533 else if (a == "-W" && i + 1 < args.size()) {
1534 std::string s = args[++i];
1535 std::string cur;
1536 for (char c : s) {
1537 if (c == ' ' || c == '\t' || c == '\n') {
1538 if (!cur.empty()) { spec.words.push_back(std::move(cur)); cur.clear(); }
1539 } else cur.push_back(c);
1540 }
1541 if (!cur.empty()) spec.words.push_back(std::move(cur));
1542 }
1543 else if (a == "-F" && i + 1 < args.size()) spec.function = args[++i];
1544 else if (a == "-C" && i + 1 < args.size()) spec.command = args[++i];
1545 else if (a == "-f") spec.include_files = true;
1546 else if (a == "-d") spec.include_dirs = true;
1547 else if (a == "-o" && i + 1 < args.size()) {
1548 const std::string& opt = args[++i];
1549 if (opt == "default") spec.default_fallback = true;
1550 else if (opt == "plusdirs") spec.plusdirs = true;
1551 else if (opt == "nospace") spec.nospace = true;
1552 }
1553 else if (!a.empty() && a[0] == '-' && a != "-" && a != "--") {
1554 // other flags — accept silently
1555 }
1556 else if (a == "--") { /* end of opts */ }
1557 else commands.push_back(a);
1558 }
1559 if (print_mode) {
1560 const auto& specs = exec.completionSpecs();
1561 if (commands.empty()) {
1562 for (const auto& kv : specs) {
1563 std::printf("complete %s\n", kv.first.c_str());
1564 }
1565 return 0;
1566 }
1567 int rc = 0;
1568 for (const auto& c : commands) {
1569 if (specs.count(c) == 0) {
1570 std::fprintf(stderr,
1571 "wbsh: complete: %s: no completion specification\n",
1572 c.c_str());
1573 rc = 1;
1574 } else {
1575 std::printf("complete %s\n", c.c_str());
1576 }
1577 }
1578 return rc;
1579 }
1580 if (remove_mode) {
1581 if (commands.empty()) {
1582 for (const auto& kv : exec.completionSpecs())
1583 exec.removeCompletionSpec(kv.first);
1584 return 0;
1585 }
1586 for (const auto& c : commands) exec.removeCompletionSpec(c);
1587 return 0;
1588 }
1589 if (commands.empty()) return 0;
1590 for (const auto& c : commands) {
1591 exec.setCompletionSpec(c, spec);
1592 }
1593 return 0;
1594 }

1617 static int builtin_umask(Executor& exec, const std::vector<std::string>& args) {
1618 // Windows has no real umask. Track a stored value in env so
1619 // scripts that read/write umask see consistent values, and
1620 // best-effort apply it via _umask (limited mask support).
1621 if (args.empty() || (args.size() == 1 && args[0] == "-S")) {
1622 std::string cur = exec.env().get("_WBSH_UMASK");
1623 if (cur.empty()) cur = "0022";
1624 if (!args.empty() && args[0] == "-S") {
1625 int v = std::stoi(cur, nullptr, 8);
1626 auto bits = [&](int shift) {
1627 std::string s;
1628 s.push_back((v & (0400 >> shift)) ? '-' : 'r');
1629 s.push_back((v & (0200 >> shift)) ? '-' : 'w');
1630 s.push_back((v & (0100 >> shift)) ? '-' : 'x');
1631 return s;
1632 };
1633 std::printf("u=%s,g=%s,o=%s\n",
1634 bits(0).c_str(), bits(3).c_str(), bits(6).c_str());
1635 } else {
1636 std::printf("%s\n", cur.c_str());
1637 }
1638 return 0;
1639 }
1640 if (args.size() == 1 && !args[0].empty() && args[0][0] != '-') {
1641 int v = 0;
1642 try { v = std::stoi(args[0], nullptr, 8); } catch (...) { return 1; }
1643 char buf[16];
1644 std::snprintf(buf, sizeof(buf), "%04o", v & 0777);
1645 exec.env().set("_WBSH_UMASK", buf);
1646#ifdef _WIN32
1647 _umask(v & 0700);
1648#endif
1649 return 0;
1650 }
1651 return 1;
1652 }
