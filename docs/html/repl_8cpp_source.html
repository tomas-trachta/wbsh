140 // Resolve the current git branch, walking up from cwd looking for .git.
141 // Returns the branch name (e.g. "main"), a short commit hash for detached
142 // HEAD, or empty if not inside a repository. Cheap enough to call every
143 // prompt redraw — bounded directory walk + one tiny file read.
144 static std::string detectGitBranch() {
145 namespace fs = std::filesystem;
146 std::error_code ec;
147 fs::path cur = fs::current_path(ec);
148 if (ec) return {};
149
150 fs::path git_dir;
151 fs::path probe = cur;
152 while (true) {
153 fs::path candidate = probe / ".git";
154 if (fs::exists(candidate, ec)) {
155 git_dir = candidate;
156 break;
157 }
158 fs::path parent = probe.parent_path();
159 if (parent == probe) break;
160 probe = parent;
161 }
162 if (git_dir.empty()) return {};
163
164 // `.git` may be a regular directory OR a file containing
165 // `gitdir: <relative-or-absolute-path>` (worktrees, submodules).
166 if (!fs::is_directory(git_dir, ec)) {
167 std::ifstream f(git_dir);
168 std::string line;
169 if (!std::getline(f, line)) return {};
170 const std::string prefix = "gitdir: ";
171 if (line.compare(0, prefix.size(), prefix) != 0) return {};
172 fs::path g(line.substr(prefix.size()));
173 if (!g.is_absolute()) g = (probe / g).lexically_normal();
174 git_dir = g;
175 }
176
177 std::ifstream head(git_dir / "HEAD");
178 std::string line;
179 if (!std::getline(head, line)) return {};
180 // Trim trailing CR/LF.
181 while (!line.empty() && (line.back() == '\r' || line.back() == '\n'))
182 line.pop_back();
183
184 const std::string ref_prefix = "ref: ";
185 if (line.compare(0, ref_prefix.size(), ref_prefix) == 0) {
186 std::string ref = line.substr(ref_prefix.size());
187 const std::string heads = "refs/heads/";
188 if (ref.compare(0, heads.size(), heads) == 0) {
189 return ref.substr(heads.size());
190 }
191 return ref;
192 }
193 // Detached HEAD: line is a 40-char SHA. Show short form.
194 if (line.size() >= 7) line.resize(7);
195 return line + " (detached)";
196 }

198 static std::string expandPrompt(const std::string& ps, Environment& env, const PathConv& pc) {
199 auto cwdStr = []() -> std::string {
200 std::error_code ec;
201 auto p = std::filesystem::current_path(ec);
202 return ec ? std::string(".") : pathToUtf8(p);
203 };
204 std::string out;
205 for (std::size_t i = 0; i < ps.size(); ++i) {
206 if (ps[i] != '\\' || i + 1 >= ps.size()) {
207 out.push_back(ps[i]);
208 continue;
209 }
210 char nx = ps[++i];
211 switch (nx) {
212 case 'n': out.push_back('\n'); break;
213 case 'r': out.push_back('\r'); break;
214 case 'a': out.push_back('\a'); break;
215 case 'e': out.push_back('\x1b'); break;
216 case '\\': out.push_back('\\'); break;
217 case '$': out.push_back('$'); break;
218 case 's': out += "wbsh"; break;
219 // \‍[ and \‍] mark non-printing regions for line-editor width
220 // accounting. We don't have a fancy editor, so just drop them.
221 case '[': break;
222 case ']': break;
223 case 'u': {
224 std::string u = env.get("USER");
225 if (u.empty()) u = env.get("USERNAME");
226 out += u;
227 break;
228 }
229 case 'h':
230 case 'H': {
231 std::string h = env.get("HOSTNAME");
232 if (h.empty()) h = env.get("COMPUTERNAME");
233 if (nx == 'h') {
234 auto dot = h.find('.');
235 if (dot != std::string::npos) h.resize(dot);
236 }
237 out += h;
238 break;
239 }
240 case 'w': {
241 std::string posix = pc.toPosix(cwdStr());
242 std::string home = env.get("HOME");
243 if (!home.empty() && posix.size() >= home.size()
244 && posix.compare(0, home.size(), home) == 0
245 && (posix.size() == home.size() || posix[home.size()] == '/')) {
246 posix = "~" + posix.substr(home.size());
247 }
248 out += posix;
249 break;
250 }
251 case 'W': {
252 std::string posix = pc.toPosix(cwdStr());
253 auto slash = posix.rfind('/');
254 out += (slash == std::string::npos) ? posix : posix.substr(slash + 1);
255 break;
256 }
257 case 'g': {
258 // `\g` — git branch (or empty when not inside a repo).
259 // Renders as " (branch)" — yellow when stdout is a TTY,
260 // plain otherwise so piped/redirected output stays clean.
261 std::string br = detectGitBranch();
262 if (br.empty()) break;
263#ifdef _WIN32
264 bool tty = _isatty(_fileno(stdout)) != 0;
265#else
266 bool tty = false;
267#endif
268 if (tty) {
269 out += " \x1b[33;1m(";
270 out += br;
271 out += ")\x1b[0m";
272 } else {
273 out += " (";
274 out += br;
275 out += ")";
276 }
277 break;
278 }
279 case 't': {
280 std::time_t t = std::time(nullptr);
281 char buf[16];
282 std::strftime(buf, sizeof(buf), "%H:%M:%S", std::localtime(&t));
283 out += buf;
284 break;
285 }
286 default:
287 out.push_back('\\');
288 out.push_back(nx);
289 break;
290 }
291 }
292 return out;
293 }

295 // =====================================================================
296 // History expansion: csh-style !! / !N / !-N / !str / ^old^new
297 // =====================================================================
298
299 // Apply csh-style history expansion to `line` against `history`.
300 // !! most recent entry
301 // !N entry by 1-based number
302 // !-N N-th most recent
303 // !str most recent entry starting with str
304 // ^old^new first occurrence of `old` in last entry replaced
305 // Returns true when the line was rewritten; on output, `expanded` holds
306 // the new text the caller should run.
307 static bool expandHistory(const std::string& line,
308 const std::vector<std::string>& history,
309 std::string& expanded) {
310 expanded.clear();
311 if (line.empty()) { expanded = line; return false; }
312 // ^old^new(^) form.
313 if (line[0] == '^') {
314 if (history.empty()) { expanded = line; return false; }
315 std::size_t p1 = line.find('^', 1);
316 if (p1 == std::string::npos) { expanded = line; return false; }
317 std::size_t p2 = line.find('^', p1 + 1);
318 std::string oldp = line.substr(1, p1 - 1);
319 std::string newp = (p2 == std::string::npos)
320 ? line.substr(p1 + 1)
321 : line.substr(p1 + 1, p2 - p1 - 1);
322 std::string base = history.back();
323 auto pos = base.find(oldp);
324 if (pos == std::string::npos) { expanded = line; return false; }
325 expanded = base.substr(0, pos) + newp + base.substr(pos + oldp.size());
326 return true;
327 }
328 bool any = false;
329 for (std::size_t i = 0; i < line.size(); ++i) {
330 char c = line[i];
331 if (c != '!') { expanded.push_back(c); continue; }
332 if (i + 1 >= line.size()) { expanded.push_back(c); continue; }
333 char nx = line[i + 1];
334 // Don't expand `!=`, `!"`, `!\`, end-of-line `!`, or `!{` (rare).
335 if (nx == ' ' || nx == '\t' || nx == '\n' || nx == '='
336 || nx == '"' || nx == '\\') { expanded.push_back(c); continue; }
337 std::string sub;
338 std::size_t advance = 0;
339 if (nx == '!') {
340 if (history.empty()) { expanded.push_back(c); continue; }
341 sub = history.back();
342 advance = 2;
343 } else if (nx == '-' && i + 2 < line.size() && std::isdigit((unsigned char)line[i + 2])) {
344 std::size_t k = i + 2;
345 while (k < line.size() && std::isdigit((unsigned char)line[k])) ++k;
346 int n = std::stoi(line.substr(i + 2, k - i - 2));
347 if (n <= 0 || static_cast<std::size_t>(n) > history.size()) {
348 expanded.push_back(c); continue;
349 }
350 sub = history[history.size() - n];
351 advance = k - i;
352 } else if (std::isdigit((unsigned char)nx)) {
353 std::size_t k = i + 1;
354 while (k < line.size() && std::isdigit((unsigned char)line[k])) ++k;
355 int n = std::stoi(line.substr(i + 1, k - i - 1));
356 if (n <= 0 || static_cast<std::size_t>(n) > history.size()) {
357 expanded.push_back(c); continue;
358 }
359 sub = history[n - 1];
360 advance = k - i;
361 } else if (std::isalpha((unsigned char)nx) || nx == '_') {
362 std::size_t k = i + 1;
363 while (k < line.size()
364 && (std::isalnum((unsigned char)line[k])
365 || line[k] == '_' || line[k] == '-')) ++k;
366 std::string prefix = line.substr(i + 1, k - i - 1);
367 const std::string* match = nullptr;
368 for (auto rit = history.rbegin(); rit != history.rend(); ++rit) {
369 if (rit->compare(0, prefix.size(), prefix) == 0) { match = &*rit; break; }
370 }
371 if (!match) { expanded.push_back(c); continue; }
372 sub = *match;
373 advance = k - i;
374 } else {
375 expanded.push_back(c);
376 continue;
377 }
378 expanded += sub;
379 i += advance - 1;
380 any = true;
381 }
382 return any;
383 }

385 // =====================================================================
386 // Multi-line continuation detection. If lex/parse complains about an
387 // unterminated quote / heredoc / pipe operand, we treat the buffer as
388 // "still being typed" and re-prompt with PS2.
389 // =====================================================================
390 static bool looksIncomplete(const std::vector<LexError>& lex_errs,
391 const std::vector<ParseError>& parse_errs) {
392 for (const auto& e : lex_errs) {
393 if (e.message.find("unterminated") != std::string::npos) return true;
394 }
395 for (const auto& e : parse_errs) {
396 const auto& m = e.message;
397 if (m.find("expected `") != std::string::npos
398 || m.find("expected pipeline") != std::string::npos
399 || m.find("expected command") != std::string::npos) {
400 return true;
401 }
402 }
403 return false;
404 }

410 static void initConsoleAndSignals(ReplState& s) {
411#ifdef _WIN32
412 SetConsoleCtrlHandler(ctrlCHandler, TRUE);
413 // UTF-8 console output so the prompt and our messages render right
414 // on Windows Terminal / conhost. Input mode left as default (line
415 // + echo + processed) so basic editing works.
416 SetConsoleOutputCP(CP_UTF8);
417 // Capture sensible "cooked" modes once and re-apply them every loop
418 // iteration. Externals (git's pager, less, vim, etc.) often change
419 // console state and don't always restore it cleanly — without this
420 // the next prompt's escape codes print literally.
421 s.h_out = GetStdHandle(STD_OUTPUT_HANDLE);
422 s.h_in = GetStdHandle(STD_INPUT_HANDLE);
423 if (s.h_out != INVALID_HANDLE_VALUE && GetConsoleMode(s.h_out, &s.good_out_mode)) {
424 s.good_out_mode |= ENABLE_VIRTUAL_TERMINAL_PROCESSING;
425 s.good_out_mode |= ENABLE_PROCESSED_OUTPUT;
426 s.good_out_mode |= ENABLE_WRAP_AT_EOL_OUTPUT;
427 s.color_ok = SetConsoleMode(s.h_out, s.good_out_mode) != 0;
428 }
429 if (s.h_in != INVALID_HANDLE_VALUE && GetConsoleMode(s.h_in, &s.good_in_mode)) {
430 s.good_in_mode |= ENABLE_LINE_INPUT;
431 s.good_in_mode |= ENABLE_ECHO_INPUT;
432 s.good_in_mode |= ENABLE_PROCESSED_INPUT;
433 s.good_in_mode |= ENABLE_EXTENDED_FLAGS;
434 SetConsoleMode(s.h_in, s.good_in_mode);
435 }
436 setupConsoleWindow();
437 setupConsoleFont(s.h_out);
438#else
439 s.color_ok = true;
440#endif /* _WIN32 */
441 }

515 // Re-apply the captured "good" console modes (externals can change
516 // them), drain a pending Ctrl-C, and detect a window resize.
517 static void pumpAsyncEvents(Environment& env, Executor& exec, ReplState& s) {
518#ifdef _WIN32
519 // External processes (especially pagers — git log -> less, vim,
520 // etc.) commonly mutate console mode and don't restore it on
521 // exit. Re-apply our good modes before every prompt so the line
522 // editor and ANSI escapes work consistently.
523 if (s.h_out != INVALID_HANDLE_VALUE) SetConsoleMode(s.h_out, s.good_out_mode);
524 if (s.h_in != INVALID_HANDLE_VALUE) SetConsoleMode(s.h_in, s.good_in_mode);
525 // If a Ctrl-C arrived while no command was running, fire the
526 // INT trap (if any) and discard any partly-typed multiline.
527 if (g_ctrlc_pending.exchange(false)) {
528 s.buffer.clear();
529 s.waiting_for_more = false;
530 std::fputc('\n', stdout);
531 if (exec.hasTrap("INT")) {
532 std::string action = exec.trapAction("INT");
533 exec.executeText(action, "<trap INT>");
534 }
535 exec.setLastStatus(130); // 128 + SIGINT
536 }
537 // Window resize → update LINES/COLUMNS and fire WINCH trap.
538 if (s.h_out != INVALID_HANDLE_VALUE) {
539 CONSOLE_SCREEN_BUFFER_INFO info{};
540 if (GetConsoleScreenBufferInfo(s.h_out, &info)) {
541 int cols = info.srWindow.Right - info.srWindow.Left + 1;
542 int lines = info.srWindow.Bottom - info.srWindow.Top + 1;
543 if (cols != s.last_cols || lines != s.last_lines) {
544 env.set("COLUMNS", std::to_string(cols));
545 env.set("LINES", std::to_string(lines));
546 if (s.last_cols != 0 && exec.hasTrap("WINCH")) {
547 std::string action = exec.trapAction("WINCH");
548 exec.executeText(action, "<trap WINCH>");
549 }
550 s.last_cols = cols;
551 s.last_lines = lines;
552 }
553 }
554 }
555#else
556 (void)env; (void)exec; (void)s;
557#endif /* _WIN32 */
558 }

587 const std::vector<ParseError>& parse_errs) {

594 err_pre, err_msg, err_loc, e.loc.line, e.loc.column, err_msg,
595 e.message.c_str());
596 }
597 for (const auto& e : parse_errs) {
598 std::fprintf(stderr, "%swbsh: parse%s %s%zu:%zu:%s %s\n",
599 err_pre, err_msg, err_loc, e.loc.line, e.loc.column, err_msg,
600 e.message.c_str());
601 }

604 // Lex+parse the accumulated buffer. If incomplete, mark the state and
605 // return so the next iteration prompts with PS2. Otherwise print any
606 // errors and (when both lex/parse were clean) execute. ShellExit
607 // thrown by execute is allowed to propagate to runInteractive, which
608 // returns it as the shell's exit status.
609 static void parseAndMaybeExecute(Executor& exec, ReplState& s) {
610 Lexer lex(s.buffer);
611 auto tokens = lex.tokenize();
612 Parser parser(std::move(tokens), s.buffer);
613 auto root = parser.parseProgram();
614
615 if (looksIncomplete(lex.errors(), parser.errors())) {
616 s.waiting_for_more = true;
617 return;
618 }
619 s.waiting_for_more = false;
620
621 printLexParseErrors(lex.errors(), parser.errors());
622
623 if (root && parser.errors().empty() && lex.errors().empty()) {
624 Node* root_ptr = root.get();
625 exec.adoptAst(std::move(root));
626 exec.setSourceText(s.buffer);
627 exec.execute(*root_ptr); // may throw ShellExit
628 }
629 s.buffer.clear();
630 }
631
632 // =========================================================================
633 // runInteractive: top-level orchestrator. Reads top-down as a story —
634 // every step is a named function, the loop body is a handful of clear
635 // calls. The bookkeeping lives in ReplState.
636 // =========================================================================
637 int runInteractive() {
638 Environment env;
639 prepareEnv(env);
640 Executor exec(env);
641 absorbInheritedState(env, exec);
642
643 ReplState state;
644 initConsoleAndSignals(state);
645 initShellDefaults(env, state);
646 printBanner(state);
647 initHistFile(env, exec, state);
648
649 try {
650 sourceWbshrc(env, exec);
651 } catch (ShellExit& e) {
652 saveHistory(exec, state);
653 return e.status;
654 }
655
656 LineEditor editor(env, exec);
657 while (true) {
658 pumpAsyncEvents(env, exec, state);
659
660 std::string prompt = buildPrompt(env, exec, state);
661 std::string line;
662 if (!editor.readLine(prompt, line)) {
663 // EOF (Ctrl-D / stdin closed). If we're mid-multi-line, just
664 // discard and re-prompt; otherwise the shell exits.
665 if (!state.buffer.empty()) {
666 state.buffer.clear();
667 state.waiting_for_more = false;
668 std::fputc('\n', stdout);
669 continue;
670 }
671 std::fputc('\n', stdout);
672 exec.fireExitTrap();
673 saveHistory(exec, state);
674 return exec.lastStatus();
675 }
676
677 if (!state.waiting_for_more) maybeExpandHistory(exec.history(), line);
678 if (!line.empty()) exec.addHistoryEntry(line);
679 appendToBuffer(state, line);
680
681 try {
682 parseAndMaybeExecute(exec, state);
683 } catch (ShellExit& e) {
684 exec.fireExitTrap();
685 saveHistory(exec, state);
686 return e.status;
687 }
688 }
689 }
