35 int bits(int need) {
36 long val = bitbuf;
37 while (bitcnt < need) {
38 if (in_pos >= in_len) { err = true; return 0; }
39 val |= (long)in[in_pos++] << bitcnt;
40 bitcnt += 8;
41 }
42 bitbuf = (int)(val >> need);
43 bitcnt -= need;
44 return (int)val & ((1 << need) - 1);
45 }
46};

48struct InflateHuffman {

50 short symbol[MAXCODES] {};

53static int decode(InflateState& s, const InflateHuffman& h) {
54 int code = 0, first = 0, index = 0;
55 for (int len = 1; len <= MAXBITS; ++len) {
56 code |= s.bits(1);
57 int count = h.count[len];
58 if (code - count < first) return h.symbol[index + (code - first)];
59 index += count;
60 first += count;
61 first <<= 1;
62 code <<= 1;
63 }
64 s.err = true;
65 return -1;
66}

68static int construct(InflateHuffman& h, const short* length, int n) {
69 short offs[MAXBITS + 1];
70 for (int len = 0; len <= MAXBITS; ++len) h.count[len] = 0;
71 for (int sym = 0; sym < n; ++sym) h.count[length[sym]]++;
72 if (h.count[0] == n) return 0;
73 int left = 1;
74 for (int len = 1; len <= MAXBITS; ++len) {
75 left <<= 1;
76 left -= h.count[len];
77 if (left < 0) return left;
78 }
79 offs[1] = 0;
80 for (int len = 1; len < MAXBITS; ++len) offs[len + 1] = offs[len] + h.count[len];
81 for (int sym = 0; sym < n; ++sym) {
82 if (length[sym] != 0) h.symbol[offs[length[sym]]++] = (short)sym;
83 }
84 return left;
85}

121static int stored(InflateState& s) {
122 s.bitbuf = 0;
123 s.bitcnt = 0;
124 if (s.in_pos + 4 > s.in_len) return -1;
125 int len = s.in[s.in_pos] | (s.in[s.in_pos + 1] << 8);
126 int nlen = s.in[s.in_pos + 2] | (s.in[s.in_pos + 3] << 8);
127 s.in_pos += 4;
128 if ((len & 0xFFFF) != ((~nlen) & 0xFFFF)) return -1;
129 if (s.in_pos + len > s.in_len) return -1;
130 s.out.insert(s.out.end(), s.in + s.in_pos, s.in + s.in_pos + len);
131 s.in_pos += len;
132 return 0;
133}

135static int fixedBlock(InflateState& s) {
136 static InflateHuffman lencode, distcode;
137 static bool built = false;
138 if (!built) {
139 short lens[FIXLCODES];
140 int sym;
141 for (sym = 0; sym < 144; ++sym) lens[sym] = 8;
142 for (; sym < 256; ++sym) lens[sym] = 9;
143 for (; sym < 280; ++sym) lens[sym] = 7;
144 for (; sym < FIXLCODES; ++sym) lens[sym] = 8;
145 construct(lencode, lens, FIXLCODES);
146 for (sym = 0; sym < MAXDCODES; ++sym) lens[sym] = 5;
147 construct(distcode, lens, MAXDCODES);
148 built = true;
149 }
150 return codes(s, lencode, distcode);
151}

153static int dynamicBlock(InflateState& s) {
154 static const short order[19] = {
155 16,17,18,0,8,7,9,6,10,5,11,4,12,3,13,2,14,1,15 };
156 int nlen = s.bits(5) + 257;
157 int ndist = s.bits(5) + 1;
158 int ncode = s.bits(4) + 4;
159 if (nlen > MAXLCODES || ndist > MAXDCODES) return -1;
160 short lengths[MAXCODES] = { 0 };
161 for (int i = 0; i < ncode; ++i) lengths[order[i]] = (short)s.bits(3);
162 InflateHuffman lencode;
163 if (construct(lencode, lengths, 19) != 0) return -1;
164 int idx = 0;
165 while (idx < nlen + ndist) {
166 int sym = decode(s, lencode);
167 if (s.err) return -1;
168 if (sym < 16) {
169 lengths[idx++] = (short)sym;
170 } else if (sym == 16) {
171 if (idx == 0) return -1;
172 int len = lengths[idx - 1];
173 int n = s.bits(2) + 3;
174 while (n--) lengths[idx++] = (short)len;
175 } else if (sym == 17) {
176 int n = s.bits(3) + 3;
177 while (n--) lengths[idx++] = 0;
178 } else {
179 int n = s.bits(7) + 11;
180 while (n--) lengths[idx++] = 0;
181 }
182 }
183 if (lengths[256] == 0) return -1;
184 InflateHuffman ll;
185 if (construct(ll, lengths, nlen) != 0 && (nlen != 1 || lengths[0] != 0)) return -1;
186 InflateHuffman dc;
187 if (construct(dc, lengths + nlen, ndist) != 0
188 && (ndist != 1 || lengths[nlen] != 0)) return -1;
189 return codes(s, ll, dc);
190}

192bool inflateRaw(const std::uint8_t* in, std::size_t in_len,
193 std::vector<std::uint8_t>& out) {
194 InflateState s(in, in_len, out);
195 int last;
196 do {
197 last = s.bits(1);
198 int type = s.bits(2);
199 if (s.err) return false;
200 int rc;
201 if (type == 0) rc = stored(s);
202 else if (type == 1) rc = fixedBlock(s);
203 else if (type == 2) rc = dynamicBlock(s);
204 else return false;
205 if (rc != 0 || s.err) return false;
206 } while (!last);
207 return true;
208}
